37  /**
38   * The name for a document.  The {@code DocumentName} is an immutable structure that handles all the intricacies of file
39   * naming on various operating systems. DocumentNames have several components:
40   * <ul>
41   *     <li>{@code root} - Where in the file system the name starts (e.g C: on windows). May be empty but not null.</li>
42   *     <li>{@code dirSeparator} - the separator between name segments (e.g. "\\" on windows, "/" on linux). May not be
43   *     empty or null.</li>
44   *     <li>{@code name} - The name of the file relative to the {@code root}. May not be null. Does NOT begin with a {@code dirSeparator}</li>
45   *     <li>{@code baseName} - The name of a directory or file from which this file is reported. A DocumentName with a
46   *     {@code name} of "foo/bar/baz.txt" and a {@code baseName} of "foo" will be reported as "bar/baz.txt". May not be null.</li>
47   *     <li>{@code isCaseSensitive} - identifies if the underlying file system is case-sensitive.</li>
48   * </ul>
49   * <p>
50   *     {@code DocumentName}s are generally used to represent files on the files system. However, they are also used to represent files
51   *     within an archive. When representing a file in an archive the baseName is the name of the enclosing archive document.
52   * </p>
53   */
54  public final class DocumentName implements Comparable<DocumentName> {
55      /** The list of all roots on the file system. */
56      static final Set<String> ROOTS = new HashSet<>();
57      /** {@code True} if the file system on which we are operating is case-sensitive. */
58      public static final boolean FS_IS_CASE_SENSITIVE;
59      /** The full name for the document. */
60      private final String name;
61      /** The name of the base directory for the document. */
62      private final String baseName;
63      /** The directory separator for this document. */
64      private final String dirSeparator;
65      /** The case-sensitive flag */
66      private final boolean isCaseSensitive;
67      /** The root for the DocumentName. May be empty but not null. */
68      private final String root;
69  
70      // determine the case sensitivity of the file system we are operating on.
71      static {
72          boolean fsSensitive = true;
73          File f = null;
74          try {
75              Path p = Files.createTempDirectory("NameSet");
76              f = p.toFile();
77              fsSensitive = !new File(f, "a").equals(new File(f, "A"));
78          } catch (IOException e) {
79              fsSensitive = true;
80          } finally {
81              if (f != null) {
82                  try {
83                      FileUtils.deleteDirectory(f);
84                  } catch (IOException e) {
85                      DefaultLog.getInstance().warn("Unable to delete temporary directory: " + f, e);
86                  }
87              }
88          }
89          FS_IS_CASE_SENSITIVE = fsSensitive;
90  
91          // determine all the roots on the file system(s).
92          File[] roots = File.listRoots();
93          if (roots != null) {
94              for (File root : roots) {
95                  String name = root.getPath();
96                  ROOTS.add(name);
97              }
98          }
99  
100     }

122      * Creates a Builder from a document name. The Builder will be configured to create a clone of the DocumentName.

130     /**
131      * Builds the DocumentName from the builder.
132      * @param builder the builder to provide the values.
133      */
134     private DocumentName(final Builder builder) {
135         this.name = builder.name;
136         this.baseName = builder.baseName;
137         this.dirSeparator = builder.dirSeparator;
138         this.isCaseSensitive = builder.isCaseSensitive;
139         this.root = builder.root;
140     }
141 
142     /**
143      * Creates a new DocumentName by adding the child to the current name.
144      * @param child the child to add (must use directory separator from this document name).
145      * @return the new document name with the same {@link #baseName}, {@link #dirSeparator} and case sensitivity as
146      * this one.
147      */
148     public DocumentName resolve(final String child) {
149         List<String> parts = new ArrayList<>();
150         parts.addAll(Arrays.asList(tokenize(name)));
151         parts.addAll(Arrays.asList(tokenize(child)));
152         String newName = String.join(dirSeparator, parts);
153         return new Builder(this).setName(newName).build();
154     }

192     public String getDirectorySeparator() {
193         return dirSeparator;
194     }

196     /**
197      * Gets the portion of the name that is not part of the base name.
198      * The resulting name will always start with the directory separator.
199      * @return the portion of the name that is not part of the base name.
200      */
201     public String localized() {
202         String result = name;
203         if (result.startsWith(baseName)) {
204             result = result.substring(baseName.length());
205         }
206         if (!result.startsWith(dirSeparator)) {
207             result = dirSeparator + result;
208         }
209         return result;
210     }
211 
212     /**
213      * Gets the portion of the name that is not part of the base name.
214      * The resulting name will always start with the directory separator.
215      * @param dirSeparator The character(s) to use to separate directories in the result.
216      * @return the portion of the name that is not part of the base name.
217      */
218     public String localized(final String dirSeparator) {
219         return String.join(dirSeparator, tokenize(localized()));
220     }
221 
222     /**
223      * Tokenizes the string based on the {@link #dirSeparator} of this DocumentName.
224      * @param source the source to tokenize
225      * @return the array of tokenized strings.
226      */
227     public String[] tokenize(final String source) {
228         return source.split("\\Q" + dirSeparator + "\\E");
229     }

231     /**
232      * Gets the last segment of the name. This is the part after the last {@link #dirSeparator}..
233      * @return the last segment of the name.
234      */
235     public String getShortName() {
236         int pos = name.lastIndexOf(dirSeparator);
237         return pos == -1 ? name : name.substring(pos + 1);
238     }

240     /**
241      * Gets the case sensitivity flag.
242      * @return {@code true} if the name is case-sensitive.

244     public boolean isCaseSensitive() {
245         return isCaseSensitive;
246     }

248     /**
249      * Returns the localized file name.
250      * @return the localized file name.
251      */
252     @Override
253     public String toString() {
254         return localized();

257     @Override
258     public int compareTo(final DocumentName o) {
259         return FS_IS_CASE_SENSITIVE ? name.compareTo(o.name) : name.compareToIgnoreCase(o.name);
260     }
261 
262     @Override
263     public boolean equals(final Object o) {
264         if (this == o) {
265             return true;
266         }
267         if (o == null || getClass() != o.getClass()) {
268             return false;
269         }
270         DocumentName that = (DocumentName) o;
271         if (isCaseSensitive() == that.isCaseSensitive() &&  Objects.equals(dirSeparator, that.dirSeparator)) {
272             return isCaseSensitive ? name.equalsIgnoreCase(that.name) : name.equals(that.name);
273         }
274         return false;
275     }
276 
277     @Override
278     public int hashCode() {
279         return Objects.hash(name, dirSeparator, isCaseSensitive());
280     }

282     /**
283      * The Builder for a DocumentName.
284      */
285     public static final class Builder {
286         /** The name for the document */
287         private String name;
288         /** The base name for the document */
289         private String baseName;
290         /** The directory separator */
291         private String dirSeparator;
292         /** The case sensitivity flag */
293         private boolean isCaseSensitive;
294         /** The file system root */
295         private String root;
296 
297         /**
298          * Create with default settings.
299          */
300         private Builder() {
301             isCaseSensitive = FS_IS_CASE_SENSITIVE;
302             dirSeparator = File.separator;
303             root = "";
304         }
305 
306         /**
307          * Create based on the file provided.
308          * @param file the file to base the builder on.
309          */
310         private Builder(final File file) {
311             this();
312             setName(file);
313             isCaseSensitive = FS_IS_CASE_SENSITIVE;
314             dirSeparator = File.separator;
315         }
316 
317         /**
318          * Create a Builder that clones the specified DocumentName.
319          * @param documentName the DocumentName to clone.
320          */
321         private Builder(final DocumentName documentName) {
322             this.root = documentName.root;
323             this.name = documentName.name;
324             this.baseName = documentName.baseName;
325             this.isCaseSensitive = documentName.isCaseSensitive;
326             this.dirSeparator = documentName.dirSeparator;
327         }
328 
329         /**
330          * Verify that the builder will build a proper DocumentName.
331          */
332         private void verify() {
333             Objects.requireNonNull(name, "Name cannot be null");
334             Objects.requireNonNull(baseName, "Basename cannot be null");
335             if (name.startsWith(dirSeparator)) {
336                 name = name.substring(dirSeparator.length());
337             }
338             if (baseName.startsWith(dirSeparator)) {
339                 baseName = baseName.substring(dirSeparator.length());
340             }
341         }
342 
343         /**
344          * Sets the root for the DocumentName.
345          * @param root the root for the DocumentName.
346          * @return this.
347          */
348         public Builder setRoot(final String root) {
349             this.root = root;
350             return this;
351         }
352 
353         /**
354          * Sets the name for this DocumentName. Will reset the root to the empty string.
355          * <p>
356          *     To correctly parse the string it must either be the directory separator specified by
357          *     {@link File#separator} or must have been explicitly set by calling {@link #setDirSeparator(String)}
358          *     before making this call.
359          * </p>
360          * @param name the name for this Document name.
361          * @return this
362          */
363         public Builder setName(final String name) {
364             Pair<String, String> pair = splitRoot(name, dirSeparator);
365             if (this.root.isEmpty()) {
366                 this.root = pair.getLeft();
367             }
368             this.name = pair.getRight();
369             return this;
370         }
371 
372         /**
373          * Extracts the root/name pair from a file.
374          * @param file the file to extract the root/name pair from.
375          * @return the root/name pair.
376          */
377         static Pair<String, String> splitRoot(final File file) {
378             return splitRoot(file.getAbsolutePath(), File.separator);
379         }
380 
381         /**
382          * Extracts the root/name pair from a name string.
383          * <p>
384          *     Package private for testing.
385          * </p>
386          * @param name the name to extract the root/name pair from.
387          * @param dirSeparator the directory separator.
388          * @return the root/name pair.
389          */
390         static Pair<String, String> splitRoot(final String name, final String dirSeparator) {
391             String workingName = name;
392             String root = "";
393             for (String sysRoot : ROOTS) {
394                 if (workingName.startsWith(sysRoot)) {
395                     workingName = workingName.substring(sysRoot.length());
396                     if (!workingName.startsWith(dirSeparator)) {
397                         if (sysRoot.endsWith(dirSeparator)) {
398                             root = sysRoot.substring(0, sysRoot.length() - dirSeparator.length());
399                         }
400                         return ImmutablePair.of(root, workingName);
401                     }
402                 }
403             }
404             return ImmutablePair.of(root, workingName);
405         }
406 
407         /**
408          * Sets the builder root if it is empty.
409          * @param root the root to set the builder root to if it is empty.
410          */
411         private void setEmptyRoot(final String root) {
412             if (this.root.isEmpty()) {
413                 this.root = root;
414             }
415         }
416 
417         /**
418          * Sets the properties from the file. This method sets the {@link #root} if it is empty, and resets {@link #name},
419          * {@link #dirSeparator} and {@link #baseName}.
420          * @param file the file to set the properties from.
421          * @return this.
422          */
423         public Builder setName(final File file) {
424             Pair<String, String> pair = splitRoot(file);
425             setEmptyRoot(pair.getLeft());
426             this.name = pair.getRight();
427             this.dirSeparator = File.separator;
428             this.baseName = name;
429             if (!file.isDirectory()) {
430                 File p = file.getParentFile();
431                 if (p != null) {
432                     setBaseName(p);
433                 }
434             }
435             return this;
436         }
437 
438         /**
439          * Sets the baseName.
440          * Will set the root if it is not set.
441          * <p>
442          *     To correctly parse the string it must either be the directory separator specified by
443          *     {@link File#separator} or must have been explicitly set by calling {@link #setDirSeparator(String)}
444          *     before making this call.
445          * </p>
446          * @param baseName the basename to use.
447          * @return this.
448          */
449         public Builder setBaseName(final String baseName) {
450             Pair<String, String> pair = splitRoot(baseName, dirSeparator);
451             setEmptyRoot(pair.getLeft());
452             this.baseName = pair.getRight();
453             return this;
454         }
455 
456         /**
457          * Sets the basename from the {@link #name} of the specified DocumentName.
458          * Will set the root the baseName has the root set.
459          * @param baseName the DocumentName to set the basename from.
460          * @return this.
461          */
462         public Builder setBaseName(final DocumentName baseName) {
463             this.baseName = baseName.getName();
464             if (!baseName.getRoot().isEmpty()) {
465                 this.root = baseName.getRoot();
466             }
467             return this;
468         }
469 
470         /**
471          * Sets the basename from a File. Sets {@link #root} and the {@link #baseName}
472          * Will set the root.
473          * @param file the file to set the base name from.
474          * @return this.
475          */
476         public Builder setBaseName(final File file) {
477             Pair<String, String> pair = splitRoot(file);
478             this.root = pair.getLeft();
479             this.baseName = pair.getRight();
480             return this;
481         }
482 
483         /**
484          * Sets the directory separator.
485          * @param dirSeparator the directory separator to use.
486          * @return this.
487          */
488         public Builder setDirSeparator(final String dirSeparator) {
489             Objects.requireNonNull(dirSeparator, "Directory separator cannot be null");
490             this.dirSeparator = dirSeparator;
491             return this;
492         }
493 
494         /**
495          * Sets the {@link #isCaseSensitive} flag.
496          * @param isCaseSensitive the expected state of the flag.
497          * @return this.
498          */
499         public Builder setCaseSensitive(final boolean isCaseSensitive) {
500             this.isCaseSensitive = isCaseSensitive;
501             return this;
502         }
503 
504         /**
505          * Build a DocumentName from this builder.
506          * @return A new DocumentName.
507          */
508         public DocumentName build() {
509             verify();
510             return new DocumentName(this);
511         }
512     }
