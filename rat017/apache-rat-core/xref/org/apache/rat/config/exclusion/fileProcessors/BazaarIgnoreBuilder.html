19  package org.apache.rat.config.exclusion.fileProcessors;
20  
21  import java.util.Optional;
22  import java.util.function.Consumer;
23  
24  import org.apache.rat.config.exclusion.MatcherSet;
25  import org.apache.rat.document.DocumentName;
26  
27  import static java.lang.String.format;
28  
29  /**
30   * A processor for {@code .bzrignore} files.
31   */
32  public final class BazaarIgnoreBuilder extends AbstractFileProcessorBuilder {
33      /**
34       * Constructor.
35       */
36      public BazaarIgnoreBuilder() {
37          super(".bzrignore", "#", true);
38      }
39  
40      @Override
41      public Optional<String> modifyEntry(final Consumer<MatcherSet> matcherSetConsumer, final DocumentName baseName, final String entry) {
42          if (entry.startsWith("RE:")) {
43              String line = entry.substring("RE:".length()).trim();
44              String pattern = line.startsWith("^") ? line.substring(1) : line;
45              return Optional.of(format(REGEX_FMT, pattern));
46          }
47          return Optional.of(entry);
48      }
49  }
