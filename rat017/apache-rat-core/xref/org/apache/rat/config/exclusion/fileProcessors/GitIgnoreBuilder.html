19  package org.apache.rat.config.exclusion.fileProcessors;
20  
21  import java.io.File;
22  import java.util.Optional;
23  import java.util.function.Consumer;
24  
25  import org.apache.rat.config.exclusion.ExclusionUtils;
26  import org.apache.rat.config.exclusion.MatcherSet;
27  import org.apache.rat.config.exclusion.plexus.MatchPatterns;
28  import org.apache.rat.document.DocumentName;
29  import org.apache.rat.document.DocumentNameMatcher;
30  
31  import static org.apache.rat.config.exclusion.ExclusionUtils.NEGATION_PREFIX;
32  
33  /**
34   * Processes the {@code .gitignore} file.
35   * @see <a href='https://git-scm.com/docs/gitignore'>.gitignore documentation</a>
36   */
37  public class GitIgnoreBuilder extends AbstractFileProcessorBuilder {
38      /** The name of the file we read from */
39      private static final String IGNORE_FILE = ".gitignore";
40      /** The comment prefix */
41      private static final String COMMENT_PREFIX = "#";
42      /** An escaped comment in the .gitignore file.  (Not a comment) */
43      private static final String ESCAPED_COMMENT = "\\#";
44      /** An escaped negation in the .gitignore file. (Not a negation) */
45      private static final String ESCAPED_NEGATION = "\\!";
46      /** The slash string */
47      private static final String SLASH = "/";
48  
49      /**
50       * Constructs a file processor that processes a {@code .gitignore} file and ignores any lines starting with {@value #COMMENT_PREFIX}.
51       */
52      public GitIgnoreBuilder() {
53          super(IGNORE_FILE, COMMENT_PREFIX, true);
54      }
55  
56      /**
57       * Convert the string entry.
58       * If the string ends with a slash an {@link DocumentNameMatcher#and} is constructed from a directory check and the file
59       * name matcher.  In this case an empty Optional is returned.
60       * If the string starts with {@value ExclusionUtils#NEGATION_PREFIX} then the entry is placed in the include list, otherwise
61       * the entry is placed in the exclude list and the name of the check returned.
62       * @param documentName The name of the document being processed.
63       * @param entry The entry from the document
64       * @return and Optional containing the name of the matcher.
65       */
66      @Override
67      protected Optional<String> modifyEntry(final Consumer<MatcherSet> matcherSetConsumer, final DocumentName documentName, final String entry) {
68          // An optional prefix "!" which negates the pattern;
69          boolean prefix = entry.startsWith(NEGATION_PREFIX);
70          String pattern = prefix || entry.startsWith(ESCAPED_COMMENT) || entry.startsWith(ESCAPED_NEGATION) ?
71                  entry.substring(1) : entry;
72  
73          // If there is a separator at the beginning or middle (or both) of the pattern, then
74          // the pattern is relative to the directory level of the particular .gitignore file itself.
75          // Otherwise, the pattern may also match at any level below the .gitignore level.
76          int slashPos = pattern.indexOf(SLASH);
77          // no slash or at end already
78          if (slashPos == -1 || slashPos == pattern.length() - 1) {
79              pattern = "**/" + pattern;
80          }
81          if (slashPos == 0) {
82              pattern = pattern.substring(1);
83          }
84          // If there is a separator at the end of the pattern then the pattern will only match directories,
85          // otherwise the pattern can match both files and directories.
86          if (pattern.endsWith(SLASH)) {
87              pattern = pattern.substring(0, pattern.length() - 1);
88              String name = prefix ? NEGATION_PREFIX + pattern : pattern;
89              DocumentName matcherPattern = DocumentName.builder(documentName).setName(name.replace(SLASH, documentName.getDirectorySeparator()))
90                      .build();
91              DocumentNameMatcher matcher = DocumentNameMatcher.and(new DocumentNameMatcher("isDirectory", File::isDirectory),
92                      new DocumentNameMatcher(name, MatchPatterns.from(matcherPattern.localized(documentName.getDirectorySeparator()))));
93  
94              MatcherSet.Builder builder = new MatcherSet.Builder();
95              if (prefix) {
96                  builder.addIncluded(matcher);
97              } else {
98                  builder.addExcluded(matcher);
99              }
100             matcherSetConsumer.accept(builder.build());
101             return Optional.empty();
102         }
103         return Optional.of(prefix ? NEGATION_PREFIX + pattern : pattern);
104     }
105 }
