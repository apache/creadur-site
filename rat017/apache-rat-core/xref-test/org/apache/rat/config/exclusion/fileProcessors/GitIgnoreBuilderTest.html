43      @Test
44      public void processExampleFileTest() throws IOException {
45          try {
46              String[] lines = {
47                      "# somethings",
48                      "!thingone", "thing*", System.lineSeparator(),
49                      "# some fish",
50                      "**/fish", "*_fish",
51                      "# some colorful directories",
52                      "red/", "blue/*/"};
53              List<String> matches = Arrays.asList("some/things", "some/fish", "another/red_fish");
54  
55              List<String> notMatches = Arrays.asList("some/thingone", "thingone");
56  
57              writeFile(".gitignore", Arrays.asList(lines));
58  
59              assertCorrect(new GitIgnoreBuilder() {
60                  @Override
61                  protected Optional<File> globalGitIgnore() {
62                      return Optional.empty();
63                  }
64              }, matches, notMatches);
65          } finally {
66              System.getProperties().remove("FSInfo");
67          }
68      }

70      // see https://git-scm.com/docs/gitignore
71      @ParameterizedTest
72      @MethodSource("modifyEntryData")
73      public void modifyEntryTest(DocumentName.FSInfo fsInfo, String source, String expected) {
74          GitIgnoreBuilder underTest = new GitIgnoreBuilder();
75          DocumentName testName = DocumentName.builder(fsInfo).setName("GitIgnoreBuilderTest").setBaseName("testDir").build();
76          List<MatcherSet> matcherSets = new ArrayList<>();
77          Optional<String> entry = underTest.modifyEntry(matcherSets::add, testName, source);
78  
79          if (source.endsWith("/")) {
80              assertThat(entry).isNotPresent();
81              assertThat(matcherSets).hasSize(1);
82              DocumentNameMatcher matcher = matcherSets.get(0).createMatcher();
83              assertThat(matcher.toString()).isEqualTo(expected);
84          } else {
85              assertThat(entry.get()).isEqualTo(expected);
86          }
87      }

89      private static Stream<Arguments> modifyEntryData() {
90          List<Arguments> lst = new ArrayList<>();
91          for (DocumentName.FSInfo fsInfo : FSInfoTest.TEST_SUITE) {
92              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
93  
94              lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
95              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
96              lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
97              lst.add(Arguments.of(fsInfo, "/filename", "filename"));
98              lst.add(Arguments.of(fsInfo, "file/name", "file/name"));
99              lst.add(Arguments.of(fsInfo, "/file/name", "file/name"));
100             lst.add(Arguments.of(fsInfo, "filename", "**/filename"));
101             lst.add(Arguments.of(fsInfo, "filename/", "not(and(isDirectory, **/filename))"));
102             lst.add(Arguments.of(fsInfo, "/filename/", "not(and(isDirectory, filename))"));
103             // inclusion by itself becomes nothing.
104             lst.add(Arguments.of(fsInfo, "!filename/", "TRUE"));
105         }
106         return lst.stream();
107     }

109     private void assertMatches(DocumentName documentName, DocumentNameMatcher matcher, String[] matching, String[] notMatching) {
110         for (String test : matching) {
111             DocumentName name = documentName.resolve(test);
112             assertThat(matcher.matches(name)).as(test).isTrue();
113         }
114         for (String test : notMatching) {
115             DocumentName name = documentName.resolve(test);
116             assertThat(matcher.matches(name)).as(test).isFalse();
117         }
118     }

120     @Test
121     public void test_RAT_335() {
122         GitIgnoreBuilder underTest = new GitIgnoreBuilder() {
123             @Override
124             protected Optional<File> globalGitIgnore() {
125                 return Optional.empty();
126             }
127         };
128         URL url = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/src/");
129         File file = new File(url.getFile());
130 
131         DocumentName documentName = DocumentName.builder(file).build();
132         List<MatcherSet> matcherSets = underTest.build(documentName);
133         DocumentNameMatcher matcher = MatcherSet.merge(matcherSets).createMatcher();
134 
135         assertThat(matcher.toString()).isEqualTo("matcherSet(or('included dir1/.gitignore', 'included .gitignore'), or('excluded dir1/.gitignore', **/.gitignore, 'excluded .gitignore'))");
136 
137         // files that should be checked:
138         List<String> notMatching = Arrays.asList("README.txt", "dir1/dir1.md", "dir2/dir2.txt", "dir3/file3.log", "dir1/file1.log", "local-should-precede-global.xml");
139 
140         // files that should be ignored:
141         List<String> matching = Arrays.asList(".gitignore", "root.md", "dir1/.gitignore", "dir1/dir1.txt",  "dir2/dir2.md", "dir3/dir3.log", "local-should-precede-global.md");
142 
143         assertCorrect(matcherSets, documentName.getBaseDocumentName(), matching, notMatching);
144     }

146     /**
147      * Test that exclusions from a global gitignore are also applied
148      *
149      * https://issues.apache.org/jira/browse/RAT-473
150      */
151     @Test
152     public void test_global_gitignore() {
153         GitIgnoreBuilder underTest = new GitIgnoreBuilder() {
154             @Override
155             protected Optional<File> globalGitIgnore() {
156                 URL globalGitIgnoreUrl = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/global-gitignore");
157                 String globalGitIgnore = globalGitIgnoreUrl.getFile();
158 
159                 return Optional.of(new File(globalGitIgnore));
160             }
161         };
162         URL url = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/src/");
163         File file = new File(url.getFile());
164 
165         DocumentName documentName = DocumentName.builder(file).build();
166         List<MatcherSet> matcherSets = underTest.build(documentName);
167         DocumentNameMatcher matcher = MatcherSet.merge(matcherSets).createMatcher();
168 
169         assertThat(matcher.toString()).isEqualTo("matcherSet(or('included dir1/.gitignore', 'included .gitignore', 'included global gitignore'), or('excluded dir1/.gitignore', **/.gitignore, 'excluded .gitignore', 'excluded global gitignore'))");
170 
171         // files that should be checked:
172         // "local-should-precede-global.md" should be 'matching'
173         // here, but that is a future improvement (RAT-476)
174         List<String> notMatching = Arrays.asList("dir1/dir1.md", "dir2/dir2.txt", "dir3/file3.log", "dir1/file1.log", "local-should-precede-global.md", "local-should-precede-global.xml");
175 
176         // files that should be ignored:
177         List<String> matching = Arrays.asList(".gitignore", "README.txt", "root.md", "dir1/.gitignore", "dir1/dir1.txt", "dir2/dir2.md", "dir3/dir3.log");
178 
179         assertCorrect(matcherSets, documentName.getBaseDocumentName(), matching, notMatching);
180     }
