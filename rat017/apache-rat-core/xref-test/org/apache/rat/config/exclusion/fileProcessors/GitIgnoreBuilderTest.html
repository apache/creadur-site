43      @Test
44      public void processExampleFileTest() throws IOException {
45          try {
46              String[] lines = {
47                      "# somethings",
48                      "!thingone", "thing*", System.lineSeparator(),
49                      "# some fish",
50                      "**/fish", "*_fish",
51                      "# some colorful directories",
52                      "red/", "blue/*/"};
53              List<String> matches = Arrays.asList("some/things", "some/fish", "another/red_fish");
54  
55              List<String> notMatches = Arrays.asList("some/thingone", "thingone");
56  
57              writeFile(".gitignore", Arrays.asList(lines));
58  
59              assertCorrect(new GitIgnoreBuilder(), matches, notMatches);
60          } finally {
61              System.getProperties().remove("FSInfo");
62          }
63      }

65      // see https://git-scm.com/docs/gitignore
66      @ParameterizedTest
67      @MethodSource("modifyEntryData")
68      public void modifyEntryTest(DocumentName.FSInfo fsInfo, String source, String expected) {
69          GitIgnoreBuilder underTest = new GitIgnoreBuilder();
70          DocumentName testName = DocumentName.builder(fsInfo).setName("GitIgnoreBuilderTest").setBaseName("testDir").build();
71          List<MatcherSet> matcherSets = new ArrayList<>();
72          Optional<String> entry = underTest.modifyEntry(matcherSets::add, testName, source);
73  
74          if (source.endsWith("/")) {
75              assertThat(entry).isNotPresent();
76              assertThat(matcherSets).hasSize(1);
77              DocumentNameMatcher matcher = matcherSets.get(0).createMatcher();
78              assertThat(matcher.toString()).isEqualTo(expected);
79          } else {
80              assertThat(entry.get()).isEqualTo(expected);
81          }
82      }
83  
84      private static Stream<Arguments> modifyEntryData() {
85          List<Arguments> lst = new ArrayList<>();
86          for (DocumentName.FSInfo fsInfo : FSInfoTest.TEST_SUITE) {
87              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
88  
89              lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
90              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
91              lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
92              lst.add(Arguments.of(fsInfo, "/filename", "filename"));
93              lst.add(Arguments.of(fsInfo, "file/name", "file/name"));
94              lst.add(Arguments.of(fsInfo, "/file/name", "file/name"));
95              lst.add(Arguments.of(fsInfo, "filename", "**/filename"));
96              lst.add(Arguments.of(fsInfo, "filename/", "not(and(isDirectory, **/filename))"));
97              lst.add(Arguments.of(fsInfo, "/filename/", "not(and(isDirectory, filename))"));
98              // inclusion by itself becomes nothing.
99              lst.add(Arguments.of(fsInfo, "!filename/", "TRUE"));
100         }
101         return lst.stream();
102     }

104     private void assertMatches(DocumentName documentName, DocumentNameMatcher matcher, String[] matching, String[] notMatching) {
105         for (String test : matching) {
106             DocumentName name = documentName.resolve(test);
107             assertThat(matcher.matches(name)).as(test).isTrue();
108         }
109         for (String test: notMatching) {
110             DocumentName name = documentName.resolve(test);
111             assertThat(matcher.matches(name)).as(test).isFalse();
112         }
113     }

115     @Test
116     public void test_RAT_335() {
117         GitIgnoreBuilder underTest = new GitIgnoreBuilder();
118         URL url = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/src/");
119         File file = new File(url.getFile());
120 
121         DocumentName documentName = DocumentName.builder(file).build();
122         List<MatcherSet> matcherSets = underTest.build(documentName);
123         DocumentNameMatcher matcher = MatcherSet.merge(matcherSets).createMatcher();
124 
125         DocumentName candidate = DocumentName.builder()
126                 .setName("/home/claude/apache/creadur-rat/apache-rat-core/target/test-classes/GitIgnoreBuilderTest/src/dir1/file1.log")
127                 .setBaseName("home/claude/apache/creadur-rat/apache-rat-core/target/test-classes/GitIgnoreBuilderTest/src/").build();
128         System.out.println("Decomposition for "+candidate);
129 
130         assertThat(matcher.toString()).isEqualTo("matcherSet(or('included dir1/.gitignore', 'included .gitignore'), or('excluded dir1/.gitignore', **/.gitignore, 'excluded .gitignore'))");
131 
132         List<String> notMatching = Arrays.asList("README.txt", "dir1/dir1.md", "dir2/dir2.txt", "dir3/file3.log", "dir1/file1.log");
133 
134         List<String> matching = Arrays.asList(".gitignore", "root.md", "dir1/.gitignore", "dir1/dir1.txt",  "dir2/dir2.md", "dir3/dir3.log");
135 
136         assertCorrect(matcherSets, documentName.getBaseDocumentName(), matching, notMatching);
137     }
