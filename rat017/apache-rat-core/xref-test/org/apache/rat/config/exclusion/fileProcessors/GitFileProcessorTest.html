43      @Test
44      public void processExampleFileTest() throws IOException {
45          String[] lines = {
46                  "# somethings",
47                  "!thingone", "thing*", System.lineSeparator(),
48                  "# some fish",
49                  "**/fish", "*_fish",
50                  "# some colorful directories",
51                  "red/", "blue/*/"};
52  
53          List<String> expected = ExtendedIterator.create(Arrays.asList("**/thing*", "**/fish", "**/*_fish").iterator())
54                  .map(s -> new File(baseDir, s).getPath()).addTo(new ArrayList<>());
55          expected.add(0, "!"+new File(baseDir, "**/thingone").getPath());
56          // "thingone",
57          writeFile(".gitignore", Arrays.asList(lines));
58  
59          GitFileProcessor processor = new GitFileProcessor();
60          List<String> actual = processor.apply(baseName);
61          assertThat(actual).isEqualTo(expected);
62  
63          actual.clear();
64          processor.customDocumentNameMatchers().forEach(x -> actual.add(x.toString()));
65          expected.clear();
66          ExtendedIterator.create(Arrays.asList("**/red", "blue/*").iterator())
67                  .map(s -> String.format("and(isDirectory, %s)", s))
68                          .forEachRemaining(expected::add);
69          assertThat(actual).isEqualTo(expected);
70      }

72      // see https://git-scm.com/docs/gitignore
73      @ParameterizedTest
74      @MethodSource("modifyEntryData")
75      public void modifyEntryTest(String source, String expected) {
76          GitFileProcessor underTest = new GitFileProcessor();
77          DocumentName testName = DocumentName.builder().setName("GitFileProcessorTest").setBaseName("testDir").build();
78          if (source.endsWith("/")) {
79              assertThat(underTest.modifyEntry(testName, source)).isEqualTo(null);
80              Iterator<DocumentNameMatcher> iter = underTest.customDocumentNameMatchers().iterator();
81              assertThat(iter).hasNext();
82              assertThat(iter.next().toString()).isEqualTo(String.format("and(isDirectory, %s)", expected));
83          } else {
84              assertThat(underTest.modifyEntry(testName, source)).isEqualTo(expected);
85              assertThat(underTest.customDocumentNameMatchers().iterator().hasNext()).isFalse();
86          }
87      }
88  
89      private static Stream<Arguments> modifyEntryData() {
90          List<Arguments> lst = new ArrayList<>();
91  
92          lst.add(Arguments.of("\\#filename", "**/#filename"));
93  
94          lst.add(Arguments.of("!#filename", "!**/#filename"));
95          lst.add(Arguments.of("\\#filename", "**/#filename"));
96          lst.add(Arguments.of("!#filename", "!**/#filename"));
97          lst.add(Arguments.of("/filename", "filename"));
98          lst.add(Arguments.of("file/name", "file/name"));
99          lst.add(Arguments.of("/file/name", "file/name"));
100         lst.add(Arguments.of("filename", "**/filename"));
101         lst.add(Arguments.of("filename/", "**/filename"));
102         lst.add(Arguments.of("/filename/", "filename"));
103 
104         return lst.stream();
105     }
