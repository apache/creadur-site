188     private void verifyAllMethodsDefinedAndNeeded(final Collection<String> unsupportedArgs) {
189         // verify all options have functions.
190         final List<String> argNames = new ArrayList<>();
191         Arg.getOptions().getOptions().forEach(o -> {
192             if (o.getLongOpt() != null) {
193                 argNames.add(o.getLongOpt());
194             }
195         });
196         argNames.removeAll(unsupportedArgs);
197         argNames.removeAll(testMap.keySet());
198         if (!argNames.isEmpty()) {
199             fail("Missing methods for: " + String.join(", ", argNames));
200         }
201 
202         // verify all functions have options.
203         argNames.clear();
204         argNames.addAll(testMap.keySet());
205         argNames.remove("help");
206         Arg.getOptions().getOptions().forEach(o -> {
207             if (o.getLongOpt() != null) {
208                 argNames.remove(o.getLongOpt());
209             }
210         });
211         if (!argNames.isEmpty()) {
212             fail("Extra methods defined: " + String.join(", ", argNames));
213         }
214         unsupportedArgs.forEach(testMap::remove);
215     }

255     private String dump(final DocumentNameMatcher nameMatcher, final DocumentName name) {
256         StringBuilder sb = new StringBuilder();
257         nameMatcher.decompose(name).forEach(s -> sb.append(s).append("\n"));
258         return sb.toString();
259     }

261     private String dump(final Option option, final String fname, final DocumentNameMatcher matcher, final DocumentName name) {
262         return String.format("Argument and Name: %s%nMatcher decomposition:%n%s", displayArgAndName(option, fname),
263                 DocumentNameMatcherTest.processDecompose(matcher, name));
264     }

266     // exclude tests
267     private void execExcludeTest(final Option option, final String[] args) {
268         String[] notExcluded = {"notbaz", "well._afile"};
269         String[] excluded = {"some.foo", "B.bar", "justbaz"};
270         try {
271             ReportConfiguration config = generateConfig(ImmutablePair.of(option, args));
272             DocumentNameMatcher excluder = config.getDocumentExcluder(baseName());
273             for (String fname : notExcluded) {
274                 DocumentName docName = mkDocName(fname);
275                 assertThat(excluder.matches(docName)).as(() -> dump(option, fname, excluder, docName)).isTrue();
276             }
277             for (String fname : excluded) {
278                 DocumentName docName = mkDocName(fname);
279                 assertThat(excluder.matches(docName)).as(() -> dump(option, fname, excluder, docName)).isFalse();
280             }
281         } catch (IOException e) {
282             fail(e.getMessage());
283         }
284     }

940     @Override
941     final public Stream<? extends Arguments> provideArguments(final ExtensionContext context) {
942         List<Arguments> lst = new ArrayList<>();
943         List<String> missingTests = new ArrayList<>();
944 
945         for (String key : OptionsList.getKeys()) {
946             OptionCollectionTest.OptionTest test = testMap.get(key);
947             if (test == null) {
948                 missingTests.add(key);
949             } else {
950                 lst.add(Arguments.of(key, test));
951             }
952         }
953         if (!missingTests.isEmpty()) {
954             System.out.println("The following tests are excluded: '" + String.join("', '", missingTests) + "'");
955         }
956         return lst.stream();
957     }
