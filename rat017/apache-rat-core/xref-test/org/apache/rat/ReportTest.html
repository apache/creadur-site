58  /**
59   * Integration tests for the {@link Report} class.
60   * Integration tests must have a specific structure.
61   * <ul>
62   *     <li>They are in directories under {@code src/it/resources/ReportText/X} where {@code X} is the name of the test. Usually
63   *     a reference to a reported bug.</li>
64   *     <li>Within the directory is a file named {@code commandLine.txt}. Each line in the file is a single command line
65   *     token. For example "--output-style XML" would be 2 lines in the file "--output-style" and "XML"</li>
66   *     <li>Within the directory there is a subdirectory named {@code src}. This is the root of the file system for RAT to scan.</li>
67   *     <li>There may be a {@code notes.md} describing the test</li>
68   *     <li>There is a {@code verify.groovy} script. When executed:
69   *     <ul>
70   *         <li>The first parameter will be then name of the file that captured the output.</li>
71   *         <li>The second parameter will be the name of the file that captured the log.</li>
72   *         <li>Any assert that fails within the Groovy script will fail the test.</li>
73   *         <li>Any value returned from the script execution will fail the test and the returned value will be
74   *         used as the failure message.</li>
75   *      </ul>
76   *      </li>
77   *      <li>If an exception is expected when Report is run with the command line a file named {@code expected-message.txt}
78   *      must be present in the directory. It must contain text that is expected to be found within the message
79   *      associated with the exception.</li>
80   * </ul>
81   */
82  public class ReportTest {
83  
84      private String[] asArgs(final List<String> argsList) {
85          return argsList.toArray(new String[argsList.size()]);
86      }
87  
88      @ParameterizedTest(name = "{index} {0}")
89      @MethodSource("args")
90      public void integrationTest(String testName, Document commandLineDoc) throws Exception {
91          DefaultLog.getInstance().log(Log.Level.INFO, "Running test for " + testName);
92          File baseDir = new File(commandLineDoc.getName().getName()).getParentFile();
93  
94          // get the arguments
95          List<String> argsList = IOUtils.readLines(commandLineDoc.reader());
96  
97          CommandLine commandLine = DefaultParser.builder().setDeprecatedHandler(DeprecationReporter.getLogReporter())
98                  .setAllowPartialMatching(true).build().parse(Arg.getOptions(), asArgs(argsList));
99  
100         File outputFile = new File(baseDir,"output.txt");
101         if (!commandLine.hasOption(Arg.OUTPUT_FILE.option())) {
102             argsList.add(0, "--" + Arg.OUTPUT_FILE.option().getLongOpt());
103             argsList.add(1, outputFile.getAbsolutePath());
104         } else {
105             outputFile = new File(baseDir, commandLine.getOptionValue(Arg.OUTPUT_FILE.option()));
106         }
107 
108         File logFile = new File(baseDir,"log.txt");
109         FileLog fileLog = new FileLog(logFile);
110         Log oldLog = null;
111         try {
112             oldLog = DefaultLog.setInstance(fileLog);
113 
114             File src = new File(baseDir, "src");
115             if (src.isDirectory()) {
116                 argsList.add(src.getAbsolutePath());
117             }
118 
119             File expectedMsg = new File(baseDir, "expected-message.txt");
120             if (expectedMsg.exists()) {
121                 String msg = IOUtils.readLines(new FileReader(expectedMsg)).get(0).trim();
122                 assertThrows(RatDocumentAnalysisException.class, () -> Report.main(asArgs(argsList)),
123                         msg);
124             } else {
125                 Report.main(asArgs(argsList));
126             }
127         } finally {
128             DefaultLog.setInstance(oldLog);
129             fileLog.close();
130         }
131 
132         File groovyScript = new File(baseDir, "verify.groovy");
133         if (groovyScript.exists()) {
134             // call groovy expressions from Java code
135             CompilerConfiguration compilerConfiguration = new CompilerConfiguration();
136 
137             GroovyShell shell = new GroovyShell(compilerConfiguration);
138             for (String classPath : System.getProperty("java.class.path").split(File.pathSeparator)) {
139                 shell.getClassLoader().addClasspath(classPath);
140             }
141             Object value = shell.run(groovyScript, new String[]{outputFile.getAbsolutePath(), logFile.getAbsolutePath()});
142             if (value != null) {
143                 fail(String.format("%s",value));
144             }
145         }
146     }
147 
148     public static Stream<Arguments> args() throws RatException {
149         List<Arguments> results = new ArrayList<>();
150         URL url = ReportTest.class.getResource("/ReportTest");
151         String urlAsFile = url.getFile();
152         if(StringUtils.isEmpty(urlAsFile)) {
153             throw new RatException("Could not find root directory for " + url);
154         }
155 
156         File baseDir = new File(url.getFile());
157         DocumentName docName = DocumentName.builder(baseDir).build();
158         AbstractFileFilter fileFilter = new NameFileFilter("commandLine.txt", docName.isCaseSensitive() ? IOCase.SENSITIVE : IOCase.INSENSITIVE);
159         fileFilter = new OrFileFilter(fileFilter, DirectoryFileFilter.INSTANCE);
160         Document document = new FileDocument(docName, baseDir, new DocumentNameMatcher(fileFilter));
161         DirectoryWalker walker = new DirectoryWalker(document);
162         RatReport report = new RatReport() {
163             @Override
164             public void report(Document document)  {
165                 if (!document.isIgnored()) {
166                     String[] tokens = DocumentName.FSInfo.getDefault().tokenize(document.getName().localized());
167                     results.add(Arguments.of(tokens[1], document));
168                 }
169             }
170         };
171         walker.run(report);
172         return results.stream();
173     }

175     /**
176      * Log that captures output for later review.
177      */
178     public static class FileLog implements Log {
179 
180         private final PrintStream logFile;
181 
182         /**
183          * The level at which we will write messages
184          */
185         private Level level;
186 
187 
188         FileLog(File logFile) throws IOException {
189             this.logFile = new PrintStream(logFile);
190             level = Level.INFO;
191         }
192 
193         /**
194          * Sets the level.Log messages below the specified level will
195          * not be written to the log.
196          *
197          * @param level the level to use when writing messages.
198          */
199         public void setLevel(final Level level) {
200             this.level = level;
201         }
202 
203         @Override
204         public Level getLevel() {
205             return level;
206         }
207 
208         @Override
209         public void log(Level level, String msg) {
210             if (isEnabled(level)) {
211                 logFile.printf("%s: %s%n", level, msg);
212             }
213         }
214 
215         public void close() {
216             logFile.close();
217         }
218     }
