39      @Test
40      public void localizeTest() {
41          DocumentName documentName = DocumentName.builder().setName("/a/b/c")
42                  .setBaseName("/a").setDirSeparator("/").setCaseSensitive(false).build();
43          assertThat(documentName.localized()).isEqualTo("/b/c");
44          assertThat(documentName.localized("-")).isEqualTo("-b-c");
45      }

47      @ParameterizedTest(name ="{index} {0}")
48      @MethodSource("validBuilderData")
49      void validBuilderTest(String testName, DocumentName.Builder builder, String root, String name, String baseName, String dirSeparator) {
50          DocumentName underTest = builder.build();
51          assertThat(underTest.getRoot()).as(testName).isEqualTo(root);
52          assertThat(underTest.getDirectorySeparator()).as(testName).isEqualTo(dirSeparator);
53          assertThat(underTest.getName()).as(testName).isEqualTo(root+dirSeparator+name);
54          assertThat(underTest.getBaseName()).as(testName).isEqualTo(root+dirSeparator+baseName);
55      }
56  
57      private static Stream<Arguments> validBuilderData() {
58          List<Arguments> lst = new ArrayList<>();
59          File f = Files.newTemporaryFile();
60  
61          Set<String> roots = new HashSet<>();
62          File[] rootary = File.listRoots();
63          if (rootary != null) {
64              for (File root : rootary) {
65                  String name = root.getPath();
66                  roots.add(name);
67              }
68          }
69  
70          String name = f.getAbsolutePath();
71          String root = "";
72          for (String sysRoot : roots) {
73              if (name.startsWith(sysRoot)) {
74                  name = name.substring(sysRoot.length());
75                  if (sysRoot.endsWith(File.separator)) {
76                      root = sysRoot.substring(0, sysRoot.length() - File.separator.length());
77                  }
78                  break;
79              }
80          }
81  
82          File p = f.getParentFile();
83          String baseName = p.getAbsolutePath().substring(root.length());
84          if (baseName.startsWith(File.separator)) {
85              baseName = baseName.substring(File.separator.length());
86          }
87          lst.add(Arguments.of("setName(file)", DocumentName.builder().setName(f), root, name, baseName, File.separator));
88          lst.add(Arguments.of("Builder(file)", DocumentName.builder(f), root, name, baseName, File.separator));
89  
90          lst.add(Arguments.of("setName(dir)", DocumentName.builder().setName(p), root, baseName, baseName, File.separator));
91          lst.add(Arguments.of("Builder(dir)", DocumentName.builder(p), root, baseName, baseName, File.separator));
92  
93          File r = new File(root.isEmpty() ? File.separator : root);
94          lst.add(Arguments.of("setName(root)", DocumentName.builder().setName(r), root, "", "", File.separator));
95          lst.add(Arguments.of("Builder(root)", DocumentName.builder(r), root, "", "", File.separator));
96  
97          lst.add(Arguments.of("foo/bar foo", DocumentName.builder().setDirSeparator("/")
98                  .setName("/foo/bar").setBaseName("foo"), "", "foo/bar", "foo", "/"));
99          DocumentName.Builder builder = DocumentName.builder().setDirSeparator("\\").setName("\\foo\\bar").setBaseName("foo")
100                 .setRoot("C:");
101         lst.add(Arguments.of("C:\\foo\\bar foo", builder, "C:", "foo\\bar", "foo", "\\"));
102 
103         return lst.stream();
104     }

106     @Test
107     public void splitRootsTest() {
108         Set<String> preserve = new HashSet<>(DocumentName.ROOTS);
109         try {
110             DocumentName.ROOTS.clear();
111             DocumentName.ROOTS.add("C:\\");
112             Pair<String, String> result = DocumentName.Builder.splitRoot("C:\\My\\path\\to\\a\\file.txt", "\\");
113             assertThat(result.getLeft()).isEqualTo("C:");
114             assertThat(result.getRight()).isEqualTo("My\\path\\to\\a\\file.txt");
115 
116 
117             DocumentName.ROOTS.clear();
118             DocumentName.ROOTS.add("/");
119             result = DocumentName.Builder.splitRoot("/My/path/to/a/file.txt", "/");
120             assertThat(result.getLeft()).isEqualTo("");
121             assertThat(result.getRight()).isEqualTo("My/path/to/a/file.txt");
122 
123         } finally {
124             DocumentName.ROOTS.clear();
125             DocumentName.ROOTS.addAll(preserve);
126         }
127     }

129     @Test
130     public void archiveEntryNameTest() {
131         Set<String> preserve = new HashSet<>(DocumentName.ROOTS);
132         try {
133             DocumentName.ROOTS.clear();
134             DocumentName.ROOTS.add("C:\\");
135 
136             DocumentName archiveName = DocumentName.builder().setDirSeparator("\\")
137                     .setName("C:\\archives\\anArchive.zip").setBaseName("archives").build();
138             assertThat(archiveName.getRoot()).isEqualTo("C:");
139             assertThat(archiveName.getDirectorySeparator()).isEqualTo("\\");
140             assertThat(archiveName.getBaseName()).isEqualTo("C:\\archives");
141             assertThat(archiveName.getName()).isEqualTo("C:\\archives\\anArchive.zip");
142             assertThat(archiveName.localized()).isEqualTo("\\anArchive.zip");
143 
144             String entryName = "./anArchiveEntry.txt";
145             DocumentName innerName = DocumentName.builder()
146                     .setDirSeparator("/").setName(entryName)
147                     .setBaseName(".").setCaseSensitive(true).build();
148             assertThat(innerName.getRoot()).isEqualTo("");
149             assertThat(innerName.getDirectorySeparator()).isEqualTo("/");
150             assertThat(innerName.getBaseName()).isEqualTo("/.");
151             assertThat(innerName.getName()).isEqualTo("/" + entryName);
152             assertThat(innerName.localized()).isEqualTo("/anArchiveEntry.txt");
153 
154             String outerNameStr = format("%s#%s", archiveName.getName(), entryName);
155             DocumentName outerName = DocumentName.builder(archiveName).setName(outerNameStr)
156                     .setCaseSensitive(innerName.isCaseSensitive()).build();
157 
158             assertThat(outerName.getRoot()).isEqualTo("C:");
159             assertThat(outerName.getDirectorySeparator()).isEqualTo("\\");
160             assertThat(outerName.getBaseName()).isEqualTo("C:\\archives");
161             assertThat(outerName.getName()).isEqualTo("C:\\archives\\anArchive.zip#./anArchiveEntry.txt");
162             assertThat(outerName.localized()).isEqualTo("\\anArchive.zip#./anArchiveEntry.txt");
163             assertThat(outerName.localized("/")).isEqualTo("/anArchive.zip#./anArchiveEntry.txt");
164 
165             // test with directory
166             entryName = "./someDir/anArchiveEntry.txt";
167             innerName = DocumentName.builder()
168                     .setDirSeparator("/").setName(entryName)
169                     .setBaseName(".").setCaseSensitive(true).build();
170             assertThat(innerName.getRoot()).isEqualTo("");
171             assertThat(innerName.getDirectorySeparator()).isEqualTo("/");
172             assertThat(innerName.getBaseName()).isEqualTo("/.");
173             assertThat(innerName.getName()).isEqualTo("/" + entryName);
174             assertThat(innerName.localized()).isEqualTo("/someDir/anArchiveEntry.txt");
175 
176             outerNameStr = format("%s#%s", archiveName.getName(), entryName);
177             outerName = DocumentName.builder(archiveName).setName(outerNameStr)
178                     .setCaseSensitive(innerName.isCaseSensitive()).build();
179 
180             assertThat(outerName.getRoot()).isEqualTo("C:");
181             assertThat(outerName.getDirectorySeparator()).isEqualTo("\\");
182             assertThat(outerName.getBaseName()).isEqualTo("C:\\archives");
183             assertThat(outerName.getName()).isEqualTo("C:\\archives\\anArchive.zip#./someDir/anArchiveEntry.txt");
184             assertThat(outerName.localized()).isEqualTo("\\anArchive.zip#./someDir/anArchiveEntry.txt");
185             assertThat(outerName.localized("/")).isEqualTo("/anArchive.zip#./someDir/anArchiveEntry.txt");
186         } finally {
187             DocumentName.ROOTS.clear();
188             DocumentName.ROOTS.addAll(preserve);
189         }
190     }
