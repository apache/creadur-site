19  package org.apache.rat.document;
20  
21  import java.util.function.Predicate;
22  import org.apache.commons.io.filefilter.NameFileFilter;
23  import org.apache.rat.config.exclusion.plexus.MatchPatterns;
24  import org.junit.jupiter.api.Test;
25  
26  import static org.assertj.core.api.Assertions.assertThat;
27  import static org.apache.rat.document.DocumentNameMatcher.MATCHES_ALL;
28  import static org.apache.rat.document.DocumentNameMatcher.MATCHES_NONE;
29  
30  public class DocumentNameMatcherTest {
31  
32      private final static DocumentNameMatcher TRUE = new DocumentNameMatcher("T", (Predicate<DocumentName>)name -> true);
33      private final static DocumentNameMatcher FALSE = new DocumentNameMatcher("F", (Predicate<DocumentName>)name -> false);
34      private final static DocumentNameMatcher SOME = new DocumentNameMatcher("X", (Predicate<DocumentName>)name -> false);
35      private final static DocumentName testName = DocumentName.builder().setName("testName").setBaseName("/").build();
36  
37      public static String processDecompose(DocumentNameMatcher matcher, DocumentName candidate) {
38          StringBuilder sb = new StringBuilder();
39          matcher.decompose(candidate).forEach(s -> sb.append(s).append("\n"));
40          return sb.toString();
41      }
42  
43      @Test
44      public void orTest() {
45          assertThat(DocumentNameMatcher.or(TRUE, FALSE).matches(testName)).as("T,F").isTrue();
46          assertThat(DocumentNameMatcher.or(FALSE, TRUE).matches(testName)).as("F,T").isTrue();
47          assertThat(DocumentNameMatcher.or(TRUE, TRUE).matches(testName)).as("T,T").isTrue();
48          assertThat(DocumentNameMatcher.or(FALSE, FALSE).matches(testName)).as("F,F").isFalse();
49      }
50  
51      @Test
52      public void andTest() {
53          assertThat(DocumentNameMatcher.and(TRUE, FALSE).matches(testName)).as("T,F").isFalse();
54          assertThat(DocumentNameMatcher.and(FALSE, TRUE).matches(testName)).as("F,T").isFalse();
55          assertThat(DocumentNameMatcher.and(TRUE, TRUE).matches(testName)).as("T,T").isTrue();
56          assertThat(DocumentNameMatcher.and(FALSE, FALSE).matches(testName)).as("F,F").isFalse();
57      }
58  
59      @Test
60      public void matcherSetTest() {
61          assertThat(DocumentNameMatcher.matcherSet(TRUE, FALSE).matches(testName)).as("T,F").isTrue();
62          assertThat(DocumentNameMatcher.matcherSet(FALSE, TRUE).matches(testName)).as("F,T").isFalse();
63          assertThat(DocumentNameMatcher.matcherSet(TRUE, TRUE).matches(testName)).as("T,T").isTrue();
64          assertThat(DocumentNameMatcher.matcherSet(FALSE, FALSE).matches(testName)).as("F,F").isTrue();
65  
66          assertThat(DocumentNameMatcher.matcherSet(MATCHES_ALL, MATCHES_ALL)).as("All,All").isEqualTo(MATCHES_ALL);
67          assertThat(DocumentNameMatcher.matcherSet(MATCHES_ALL, MATCHES_NONE)).as("All,None").isEqualTo(MATCHES_ALL);
68          assertThat(DocumentNameMatcher.matcherSet(MATCHES_ALL, SOME)).as("All,X").isEqualTo(MATCHES_ALL);
69  
70          assertThat(DocumentNameMatcher.matcherSet(MATCHES_NONE, MATCHES_ALL)).as("None,All").isEqualTo(MATCHES_NONE);
71          assertThat(DocumentNameMatcher.matcherSet(MATCHES_NONE, MATCHES_NONE)).as("None,None").isEqualTo(MATCHES_ALL);
72          assertThat(DocumentNameMatcher.matcherSet(MATCHES_NONE, SOME).toString()).as("None,X").isEqualTo("not(X)");
73  
74          assertThat(DocumentNameMatcher.matcherSet(SOME, MATCHES_ALL).toString()).as("X,All").isEqualTo("matcherSet(X, TRUE)");
75          assertThat(DocumentNameMatcher.matcherSet(SOME, MATCHES_NONE)).as("X,None").isEqualTo(MATCHES_ALL);
76          assertThat(DocumentNameMatcher.matcherSet(SOME, SOME).toString()).as("X,X").isEqualTo("matcherSet(X, X)");
77      }
78  
79      @Test
80      void testDecompose() {
81          DocumentNameMatcher matcher1 = new DocumentNameMatcher("FileFilterTest", new NameFileFilter("File.name"));
82          String result = processDecompose(matcher1, testName);
83          assertThat(result).contains("FileFilterTest: >>false<<").contains("  NameFileFilter(File.name)");
84  
85          DocumentNameMatcher matcher2 = new DocumentNameMatcher("MatchPatternsTest", MatchPatterns.from("/", "**/test1*", "**/*Name"));
86          result = processDecompose(matcher2, testName);
87          assertThat(result).contains("MatchPatternsTest: >>true<<").contains("  **/test1*: >>false<<").contains("  **/*Name: >>true<<");
88  
89          DocumentNameMatcher matcher3 = DocumentNameMatcher.matcherSet(matcher1, matcher2);
90          result = processDecompose(matcher3, testName);
91          assertThat(result).contains("MatchPatternsTest: >>true<<").contains("  **/test1*: >>false<<").contains("  **/*Name: >>true<<");
92      }
93  }
