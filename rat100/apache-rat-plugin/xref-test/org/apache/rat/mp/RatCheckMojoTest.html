74      private RatCheckMojo getMojo(File pomFile) throws IOException {
75          try {
76              final RatCheckMojo mojo = new OptionMojoTest.SimpleMojoTestcase() {
77              }.getMojo(pomFile);
78              Assertions.assertNotNull(mojo);
79              return mojo;
80          } catch (IOException e) {
81              throw e;
82          } catch (Exception e) {
83              throw new IOException(format("Unable to generate mojo for %s", pomFile), e);
84          }
85      }

87      /**
88       * Creates a new instance of {@link AbstractRatMojo}.
89       *
90       * @param testDir The directory, where to look for a pom.xml file.
91       * copy location.
92       * @return The configured Mojo.
93       * @throws Exception An error occurred while creating the Mojo.
94       */
95      private RatCheckMojo newRatMojo(String testDir) throws Exception {
96          Arg.reset();
97          final File sourceDir = Resources.getResourceDirectory(format("unit/%s", testDir));
98          final File baseDir = tempDir.resolve(testDir).toFile();
99          FileUtils.copyDirectory(sourceDir, baseDir);
100         final File pomFile = new File(baseDir, "pom.xml");
101         RatCheckMojo mojo = getMojo(pomFile);
102         assertThat(mojo).isNotNull();
103         assertThat(mojo.getProject())
104                 .as("The mojo is missing its MavenProject, which will result in an NPE during RAT runs.")
105                 .isNotNull();
106 
107         File buildDirectory = new File(baseDir, "target");
108         assertThat(buildDirectory.mkdirs()).isTrue();
109         final File ratTxtFile = new File(buildDirectory, "rat.txt");
110         FileUtils.write(ratTxtFile, "", UTF_8); // Ensure the output file exists and is empty (rerunning the test will append)
111         mojo.setOutputFile(ratTxtFile.getAbsolutePath());
112         return mojo;
113     }

159     private static Map<String, String> mapOf(String... parts) {
160         Map<String, String> map = new HashMap<>();
161         for (int i = 0; i < parts.length; i += 2) {
162             map.put(parts[i], parts[i + 1]);
163         }
164         return map;
165     }

174         final RatCheckMojo mojo = newRatMojo("it2");
175         final File ratTxtFile = mojo.getRatTxtFile();
176         final String[] expected = {

179                 ReporterTestUtils.counterText(ClaimStatistic.Counter.BINARIES, 0, false),

182                 ReporterTestUtils.counterText(ClaimStatistic.Counter.IGNORED, 1, false),
183                 ReporterTestUtils.apacheLicenseVersion2(1),
184                 ReporterTestUtils.unknownLicense(1),
185                 ReporterTestUtils.documentOut(false, Document.Type.STANDARD, "/src.txt") +

187                 ReporterTestUtils.documentOut(true, Document.Type.STANDARD, "/pom.xml") +
188                         ReporterTestUtils.APACHE_LICENSE

190         try {
191             mojo.execute();
192             fail("Expected RatCheckException");

194             final String msg = e.getMessage();

197             assertThat(msg.toUpperCase()).contains("UNAPPROVED EXCEEDED MINIMUM");
198 
199             ensureRatReportIsCorrect(ratTxtFile, expected, TextUtils.EMPTY);
200         }

226             if (type == Document.Type.STANDARD) {
227                 XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/documentType[@name='STANDARD']", "count",
228                         "2");
229             } else if (type == Document.Type.IGNORED) {
230                 XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/documentType[@name='IGNORED']", "count",
231                         "1");
232             } else {
233                 XmlUtils.assertIsNotPresent(document, xPath, format("/rat-report/statistics/documentType[@name='%s']", type));
234             }
235         }

348         final RatCheckMojo mojo = newRatMojo("RAT-335");
349         final File ratTxtFile = mojo.getRatTxtFile();

351             mojo.execute();
352             fail("Expected RatCheckException");
353         } catch (RatCheckException e) {
354             final String msg = e.getMessage();
355             assertThat(msg).contains(ratTxtFile.getName());
356             assertThat(msg).contains("UNAPPROVED exceeded minimum");
357 
358             Map<ClaimStatistic.Counter, String> data = new HashMap<>();
359             data.put(ClaimStatistic.Counter.APPROVED, "1");
360             data.put(ClaimStatistic.Counter.ARCHIVES, "0");
361             data.put(ClaimStatistic.Counter.BINARIES, "0");
362             data.put(ClaimStatistic.Counter.DOCUMENT_TYPES, "3");
363             data.put(ClaimStatistic.Counter.IGNORED, "6");
364             data.put(ClaimStatistic.Counter.LICENSE_CATEGORIES, "2");
365             data.put(ClaimStatistic.Counter.LICENSE_NAMES, "2");
366             data.put(ClaimStatistic.Counter.NOTICES, "1");
367             data.put(ClaimStatistic.Counter.STANDARDS, "5");
368             data.put(ClaimStatistic.Counter.UNAPPROVED, "4");
369             data.put(ClaimStatistic.Counter.UNKNOWN, "4");
370 
371             org.w3c.dom.Document document = XmlUtils.toDom(Files.newInputStream(ratTxtFile.toPath()));
372 
373             for (ClaimStatistic.Counter counter : ClaimStatistic.Counter.values()) {
374                 String xpath = String.format("/rat-report/statistics/statistic[@name='%s']", counter.displayName());
375                 Map<String, String> map = mapOf("approval",
376                         Boolean.toString(counter != ClaimStatistic.Counter.UNAPPROVED),
377                         "count", data.get(counter),
378                         "description", counter.getDescription());
379                 XmlUtils.assertAttributes(document, xPath, xpath, map);
380             }
381 
382             // license categories
383             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/licenseCategory[@name='?????']",
384                     mapOf("count", "4"));
385 
386             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/licenseCategory[@name='AL   ']",
387                     mapOf("count", "1"));
388 
389             // license names
390             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/licenseName[@name='Apache License 2.0']",
391                     mapOf("count", "1"));
392 
393             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/licenseName[@name='Unknown license']",
394                     mapOf("count", "4"));
395 
396             // Document types
397             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/documentType[@name='IGNORED']",
398                     mapOf("count", "6"));
399 
400             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/documentType[@name='NOTICE']",
401                     mapOf("count", "1"));
402 
403             XmlUtils.assertAttributes(document, xPath, "/rat-report/statistics/documentType[@name='STANDARD']",
404                     mapOf("count", "5"));
405 
406             List<String> ignoredFiles = new ArrayList<>(Arrays.asList(
407                     "/dir1/dir1.txt",
408                     "/dir1/.gitignore",
409                     "/dir2/dir2.md",
410                     "/dir3/dir3.log",
411                     "/.gitignore",
412                     "/root.md"));
413 
414             NodeList nodeList = XmlUtils.getNodeList(document, xPath, "/rat-report/resource[@type='IGNORED']");
415             for (int i = 0; i < nodeList.getLength(); i++) {
416                 NamedNodeMap attr = nodeList.item(i).getAttributes();
417                 String s = attr.getNamedItem("name").getNodeValue();
418                 assertThat(ignoredFiles).contains(s);
419                 ignoredFiles.remove(s);
420             }
421             assertThat(ignoredFiles).isEmpty();
422         }
423     }
