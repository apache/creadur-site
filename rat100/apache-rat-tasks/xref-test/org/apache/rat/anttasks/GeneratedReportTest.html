103                         @Override
104                         protected String getMethodFormat(final AntOption antOption) {
105                             return format("<%1$s>%%s</%1$s>", tag);

107                     };
108             }
109             ARG_TYPE_MAP.put(argType, buildType);

164     @ParameterizedTest(name = "{index} {0}")
165     @MethodSource("generatedData")
166     void argumentTests(String name, String buildXml, AntOption option) throws IOException {
167         DefaultLog.getInstance().debug("Running " + name);
168         final File antFile = new File(tempDir.toFile(), targetName(option) + ".xml").getAbsoluteFile();
169         try (Writer writer = new OutputStreamWriter(Files.newOutputStream(antFile.toPath()))) {
170             writer.write(buildXml);
171         }
172         DocumentName documentName = DocumentName.builder(antFile).setBaseName(antFile.getParentFile()).build();
173         System.setProperty(MagicNames.PROJECT_BASEDIR, documentName.getBaseName());
174         StringBuilder fullLogBuffer = new StringBuilder();
175         StringBuilder outputBuffer = new StringBuilder();
176         StringBuilder errorBuffer = new StringBuilder();
177         DefaultLog.setInstance(null);
178         Project project = new Project();
179         if (Boolean.getBoolean(MagicTestNames.TEST_BASEDIR_IGNORE)) {
180             System.clearProperty(MagicNames.PROJECT_BASEDIR);
181         }
182         project.init();
183         project.setProperty(MagicTestNames.TEST_PROCESS_ID, ProcessUtil.getProcessId("<Process>"));
184         project.setProperty(MagicTestNames.TEST_THREAD_NAME, Thread.currentThread().getName());
185         project.setUserProperty(MagicNames.ANT_FILE, antFile.getAbsolutePath());
186         AntTestListener listener = new AntTestListener(option.getName(), fullLogBuffer, Project.MSG_DEBUG);
187         project.addBuildListener(listener);
188         ProjectHelper.configureProject(project, antFile);
189         executeTarget(outputBuffer, errorBuffer, project, targetName(option));
190         if (option.isDeprecated()) {
191             assertThat(listener.logBuffer).contains(option.getDeprecated());
192         }
193     }

195     private void executeTarget(StringBuilder outputBuffer, StringBuilder errorBuffer, Project project, String targetName) {
196         PrintStream out = new PrintStream(new AntOutputStream(outputBuffer));
197         PrintStream err = new PrintStream(new AntOutputStream(errorBuffer));
198 
199         /* we synchronize to protect our custom output streams from being overridden
200          * by other tests executing targets concurrently. Ultimately this would only
201          * happen if we ran a multi-threaded test executing multiple targets at once, and
202          * this protection doesn't prevent a target from internally modifying the output
203          * stream during a test - but at least this scenario is fairly deterministic so
204          * easier to troubleshoot.
205          */
206         synchronized (System.out) {
207             PrintStream sysOut = System.out;
208             PrintStream sysErr = System.err;
209             sysOut.flush();
210             sysErr.flush();
211             try {
212                 System.setOut(out);
213                 System.setErr(err);
214                 project.executeTarget(targetName);
215             } finally {
216                 System.setOut(sysOut);
217                 System.setErr(sysErr);
218             }
219         }
220     }

222     static String targetName(AntOption option) {

224         return actualOption.getName() + (actualOption.isAttribute() ? "Attribute" :"Element");

227     /**
228      * Generate the data for the tests.
229      * @return the arguments for the tests.
230      */
231     static Stream<Arguments> generatedData() {
232 
233         List<AntOption> options = Arg.getOptions().getOptions().stream()
234                 .filter(o -> !AntOption.getFilteredOptions().contains(o)).map(AntOption::new)
235                 .toList();
236 
237         List<Arguments> lst = new ArrayList<>();
238 
239         for (AntOption option : options) {
240             lst.add(createTest(option));
241             option.convertedFrom().forEach(o -> lst.add(createTest(new AntOption(o))));
242         }
243         for (Arguments arguments : lst) {
244             Object[] objects = arguments.get();
245             System.out.format("%s -> %s%n", objects[0], objects[1]);
246         }
247         return lst.stream();
248     }

250     private static Arguments createTest(AntOption option) {
251         AntOption actualOption = option.getActualAntOption();
252         BuildType buildType = ARG_TYPE_MAP.get(option.getArgType());
253         String xml = buildXml(actualOption, option, buildType.getXml(option));
254         return Arguments.of(buildType.testName(option), xml, option);
255     }

257     private static String buildXml(AntOption actualOption, AntOption option, String body) {
258         StringBuilder xml = new StringBuilder(BUILD_XML_PREFIX);
259         xml.append(format("<!-- %s -->%n", option.getName()));
260         xml.append(format("  <target name='%s'>%n", targetName(option)));
261         if (actualOption.isAttribute()) {
262             xml.append(format("    <rat:report %s=\"%s\"", actualOption.getName(), getData(option)));
263             String additionalAttributes = REQUIRED_ATTRIBUTES.get(actualOption.getName());
264             if (additionalAttributes != null) {
265                 xml.append(format(" %s", additionalAttributes));
266             }
267             xml.append(format(" >%n"));
268         } else {
269             xml.append("    <rat:report");
270             String additionalAttributes = REQUIRED_ATTRIBUTES.get(actualOption.getName());
271             if (additionalAttributes != null) {
272                 xml.append(format(" %s", additionalAttributes));
273             }
274             xml.append(">\n");
275             if (body == null) {
276                 xml.append(format("      <%1$s>%2$s</%1$s>%n", actualOption.getName(), getData(option)));
277             } else {
278 //                if (actualOption.argCount() == 1) {
279 //                    xml.append(format("      <%s %s=\"%s\" />%n", actualOption.getName(), createAttribute(option), getData(option)));
280 //                } else {
281                     xml.append(format("      <%1$s>%2$s</%1$s>%n", actualOption.getName(), body));
282 //                }
283             }
284         }
285 
286         String additionalElements = REQUIRED_ELEMENTS.get(option.getName());
287         if (additionalElements != null) {
288             xml.append(format("      %s%n", additionalElements));
289         }
290         xml.append(format("      <file file='test.file' />%n"));
291         xml.append(format("    </rat:report>%n"));
292         xml.append(format("  </target>%n%n</project>%n"));
293         return xml.toString();
294     }

296     private static String getData(AntOption option) {
297         String value = getData(option.getName());
298         if (value == null) {
299             if (!option.hasArg()) {
300                 return "true";
301             } else {
302                 throw new IllegalStateException("Missing " + option.getName());
303             }
304         }
305         return value;
306     }

393     private static class AntTestListener implements BuildListener {
394         private final int logLevel;
395         private final StringBuilder logBuffer;
396         private final StringBuilder fullLogBuffer;
397         /**
398          * Constructs a test listener which will ignore log events
399          * above the given level.
400          */
401         public AntTestListener(String name, StringBuilder fullLogBuffer, int logLevel) {
402             this.logBuffer = new StringBuilder();
403             this.fullLogBuffer = fullLogBuffer;
404             this.logLevel = logLevel;
405         }

456         /**
457          * Fired whenever a message is logged.
458          *
459          * @see BuildEvent#getMessage()
460          * @see BuildEvent#getPriority()
461          */
462         public void messageLogged(BuildEvent event) {
463             if (event.getPriority() > logLevel) {
464                 // ignore event
465                 return;
466             }
467             if (event.getPriority() <= Project.MSG_INFO) {
468                 logBuffer.append(format("[%s] %s%n", Report.fromProjectLevel(event.getPriority()), event.getMessage()));
469             }
470             fullLogBuffer.append(format("[%s] %s%n", Report.fromProjectLevel(event.getPriority()), event.getMessage()));
471         }

486     public static class BuildType {
487         /** The configuration tag for this build type */
488         protected final String tag;
489         /** If True adds the tag as the test extension */
490         private final boolean addExt;
491 
492         BuildType(final String tag) {
493             this(tag, StringUtils.isNotEmpty(tag));
494         }
495 
496         BuildType(final String tag, boolean addExt) {
497             this.tag = tag;
498             this.addExt = addExt;
499         }
500 
501         protected String getMultipleFormat(final AntOption antOption) {
502             return String.format("  <%1$s>%%s</%1$s>\n", tag);
503         }
504 
505         protected String getMethodFormat(final AntOption antOption) {
506             return antOption.hasArgs() ? getMultipleFormat(antOption) : String.format("  <%1$s>%%s</%1$s>\n", tag);
507         }
508 
509         public String testName(final AntOption antOption) {
510             return addExt ? format("%s_%s", antOption.getName(), antOption.getArgName()) : antOption.getName();
511         }
512 
513         public String getXml(final AntOption antOption) {
514             AntOption delegateOption = antOption.getActualAntOption();
515             if (delegateOption.isAttribute()) {
516                 return "";
517             } else {
518                 return format(getMethodFormat(antOption), getData(antOption));
519             }
520         }
