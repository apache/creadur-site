372     /**
373      * Finds a node via xpath on the document. And then checks family, approval and
374      * type of elements of the node.
375      *
376      * @param doc the document to check
377      * @param xpath the XPath instance to use.
378      * @param resource the XPath statement to locate the node.
379      * @param licenseInfo the license info for the node (can be {@code null}).
380      * @param type the type of resource located.
381      * @throws Exception on XPath error.
382      */
383     private static void checkNode(final Document doc, final XPath xpath, final String resource, final LicenseInfo licenseInfo,
384                                   final String type) throws Exception {
385         XmlUtils.getNode(doc, xpath, String.format("/rat-report/resource[@name='%s'][@type='%s']", resource, type));
386         if (licenseInfo != null) {
387             XmlUtils.getNode(doc, xpath,
388                     String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s'][@family='%s']",
389                             resource, type, licenseInfo.id, licenseInfo.family));
390             XmlUtils.getNode(doc, xpath,
391                     String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s'][@approval='%s']",
392                             resource, type, licenseInfo.id, licenseInfo.approval));
393             if (licenseInfo.hasNotes) {
394                 XmlUtils.getNode(doc, xpath,
395                         String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s']/notes", resource,
396                                 type, licenseInfo.id));
397             }
398         }
399     }

401     private ReportConfiguration initializeConfiguration() throws URISyntaxException {
402         Defaults defaults = Defaults.builder().build();
403         final File elementsFile = Resources.getExampleResource("exampleData");
404         final ReportConfiguration configuration = new ReportConfiguration();
