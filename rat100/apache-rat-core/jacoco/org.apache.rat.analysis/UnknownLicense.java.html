package org.apache.rat.analysis;

import org.apache.rat.license.ILicense;
import org.apache.rat.license.ILicenseFamily;

/**
 * An ILicense implementation that represents an unknown license.
 * <p>
 * The UnknownLicense is used during processing to report that a document
 * license can not be determined.
 */
public final class UnknownLicense implements ILicense {

    /**
     * The single instance of this class.
     */
    public static final UnknownLicense INSTANCE = new UnknownLicense();

    /**
     * The license family for the unknown license.
     */
    private final ILicenseFamily family;

    /**
     * Do not allow other constructions.
     */
    private UnknownLicense() {
        family = ILicenseFamily.UNKNOWN;
    }

    @Override
    public String getId() {
        return family.getFamilyCategory();
    }

    @Override
    public void reset() {
        // do nothing
    }

    @Override
    public boolean matches(final IHeaders headers) {
        return false;
    }

    @Override
    public boolean equals(final Object o) {
        return super.equals(o);
    }

    @Override
    public int hashCode() {
        return family.hashCode();
    }

    @Override
    public ILicenseFamily getLicenseFamily() {
        return family;
    }

    @Override
    public String getNote() {
        return null;
    }

    @Override
    public String getName() {
        return family.getFamilyName();
    }

    @Override
    public IHeaderMatcher getMatcher() {
        return null;
    }
}
