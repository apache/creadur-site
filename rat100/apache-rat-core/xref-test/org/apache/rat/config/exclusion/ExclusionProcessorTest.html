62      private void testParseExclusion(DocumentName basedir, DocumentNameMatcher nameMatcher, DocumentName name, boolean expected) {
63          assertThat(nameMatcher.matches(name)).as(() -> format("Failed on [%s %s]%n%s", basedir, name, dump(nameMatcher, name))).isEqualTo(expected);
64      }
65  
66      private String dump(DocumentNameMatcher nameMatcher, DocumentName name) {
67          StringBuilder sb = new StringBuilder();
68          nameMatcher.decompose(name).forEach(s -> sb.append(s).append("\n"));
69          return sb.toString();
70      }

72      private DocumentName mkName(DocumentName baseDir, String pth) throws IOException {
73          DocumentName result = baseDir.resolve(ExclusionUtils.convertSeparator(pth, "/", baseDir.getDirectorySeparator()));
74          DocumentName mocked = spy(result);
75  
76          String fn = result.localized(FileSystems.getDefault().getSeparator());
77          File file = tempDir.resolve(fn.substring(1)).toFile();
78          File parent = file.getParentFile();
79          if (parent.exists() && !parent.isDirectory() && !parent.delete()) {
80              fail(() -> "Unable to delete parent: " + parent);
81          }
82          if (!parent.exists() && !parent.mkdirs()) {
83              fail((() -> "Unable to create parent: " + parent));
84          }
85          if (file.exists()) {
86              if (file.isDirectory()) {
87                  FileUtils.deleteDirectory(file);
88              } else {
89                  FileUtils.delete(file);
90              }
91          }
92          if (!file.createNewFile()) {
93              fail(() -> "Unable to create file: " + file);
94          }
95          when(mocked.asFile()).thenReturn(file);
96          return mocked;
97      }

115         testParseExclusion(basedir, p.getNameMatcher(basedir),  mkName(basedir, "%hello%"), false);

132     private void assertExclusions(DocumentName basedir, String pattern, Map<String,Boolean> expectedMap) throws IOException {
133         String[] paths = {"a/b/foo", "b/foo", "foo", "foo/x", "foo/x/y", "b/foo/x",
134                 "b/foo/x/y", "a/b/foo/x", "a/b/foo/x/y"};
135         ExclusionProcessor p = new ExclusionProcessor().addExcludedPatterns(Collections.singletonList(pattern));
136         DocumentNameMatcher pathMatcher = p.getNameMatcher(basedir);
137         for (String pth : paths) {
138             Boolean expected = expectedMap.get(pth);
139             if (expected == null) {
140                 throw new RuntimeException("Missing expected value for " + pth + " in pattern " + pattern);
141             }
142             testParseExclusion(basedir, pathMatcher, mkName(basedir, pth), expected);
143         }
144     }

363     public static void assertSame(ExclusionProcessor actual, ExclusionProcessor expected) {
364         assertThat(actual.getExcludedCollections()).containsExactlyElementsOf(expected.getExcludedCollections());
365         assertThat(actual.getFileProcessors()).containsExactlyElementsOf(expected.getFileProcessors());
366         assertThat(actual.getIncludedCollections()).containsExactlyElementsOf(expected.getIncludedCollections());
367         assertThat(actual.getExcludedPatterns()).containsExactlyElementsOf(expected.getExcludedPatterns());
368         assertThat(actual.getIncludedPatterns()).containsExactlyElementsOf(expected.getIncludedPatterns());
369 
370         // Paths only match on name. Deserialized paths are not functional.
371         List<String> actualPaths = actual.getExcludedPaths().stream().map(DocumentNameMatcher::toString).toList();
372         List<String> expectedPaths = expected.getExcludedPaths().stream().map(DocumentNameMatcher::toString).toList();
373         assertThat(actualPaths).as("excluded paths").containsExactlyElementsOf(expectedPaths);
374 
375         actualPaths = actual.getIncludedPaths().stream().map(DocumentNameMatcher::toString).toList();
376         expectedPaths = expected.getIncludedPaths().stream().map(DocumentNameMatcher::toString).toList();
377         assertThat(actualPaths).as("included paths").containsExactlyElementsOf(expectedPaths);
378     }

433         ExclusionProcessor underTest = new ExclusionProcessor()
434                 .addExcludedPatterns(List.of("pattern/**", "pattern2/**"))
435                 .addExcludedCollection(StandardCollection.BAZAAR)
436                 .addExcludedCollection(StandardCollection.MISC)
437                 .addFileProcessor(StandardCollection.HIDDEN_FILE)
438                 .addExcludedMatcher(DocumentNameMatcher.MATCHES_ALL)
439                 .addIncludedPatterns(List.of("**/pattern3", "**/pattern4"))
440                 .addIncludedCollection(StandardCollection.ARCH)
441                 .addIncludedCollection(StandardCollection.BITKEEPER)
442                 .addIncludedMatcher(DocumentNameMatcher.MATCHES_NONE);
443 
444         StringWriter writer = new StringWriter();
445         underTest.reportExclusions(writer);
446         String actual = writer.toString();

448         assertThat(actual).containsPattern("Excluding patterns:[^$]+\\Qpattern/**\\E")
449                 .containsPattern("Excluding patterns:[^$]+\\Qpattern2/**\\E")

451                 .containsPattern("Including patterns:[^$]+\\Q**/pattern4\\E")
452                 .contains("Excluding " + StandardCollection.BAZAAR + " collection.")
453                 .contains("Excluding " + StandardCollection.MISC + " collection.")
454                 .contains("Including " + StandardCollection.ARCH + " collection.")
455                 .contains("Including " + StandardCollection.BITKEEPER + " collection.")
456                 .contains("Processing exclude file from " + StandardCollection.HIDDEN_FILE)
457                 .contains("Excluding " + DocumentNameMatcher.MATCHES_ALL + ".")
458                 .contains("Including " + DocumentNameMatcher.MATCHES_NONE + ".");
