99          assertThat(statistic.getCounter(ClaimStatistic.Counter.BINARIES)).isEqualTo(2);
100         assertThat(statistic.getCounter(ClaimStatistic.Counter.DOCUMENT_TYPES)).isEqualTo(5);

104         assertThat(statistic.getCounter(ClaimStatistic.Counter.NOTICES)).isEqualTo(2);
105         assertThat(statistic.getCounter(ClaimStatistic.Counter.STANDARDS)).isEqualTo(8);
106         assertThat(statistic.getCounter(ClaimStatistic.Counter.UNAPPROVED)).isEqualTo(2);

117         expected.put("The Telemanagement Forum License", 1);

304     /**
305      * Finds a node via xpath on the document. And then checks family, approval and
306      * type of elements of the node.
307      *
308      * @param doc The document to check
309      * @param xpath the XPath instance to use.
310      * @param resource the xpath statement to locate the node.
311      * @param licenseInfo the license info for the node. (can be null)
312      * @param type the type of resource located.
313      * @throws Exception on XPath error.
314      */
315     private static void checkNode(final Document doc, final XPath xpath, final String resource, final LicenseInfo licenseInfo,
316                                   final String type) throws Exception {
317         XmlUtils.getNode(doc, xpath, String.format("/rat-report/resource[@name='%s'][@type='%s']", resource, type));
318         if (licenseInfo != null) {
319             XmlUtils.getNode(doc, xpath,
320                     String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s'][@family='%s']",
321                             resource, type, licenseInfo.id, licenseInfo.family));
322             XmlUtils.getNode(doc, xpath,
323                     String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s'][@approval='%s']",
324                             resource, type, licenseInfo.id, licenseInfo.approval));
325             if (licenseInfo.hasNotes) {
326                 XmlUtils.getNode(doc, xpath,
327                         String.format("/rat-report/resource[@name='%s'][@type='%s']/license[@id='%s']/notes", resource,
328                                 type, licenseInfo.id));
329             }
330         }
331     }
