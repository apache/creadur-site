23  import java.io.PrintWriter;
24  import java.nio.charset.StandardCharsets;
25  
26  import org.apache.commons.cli.CommandLine;
27  import org.apache.commons.cli.DefaultParser;
28  import org.apache.commons.cli.Options;
29  import org.apache.commons.cli.ParseException;
30  import org.apache.rat.api.RatException;
31  import org.apache.rat.commandline.Arg;
32  import org.apache.rat.commandline.ArgumentContext;
33  import org.apache.rat.help.Licenses;
34  import org.apache.rat.report.Reportable;
35  import org.apache.rat.ui.UIOption;
36  import org.apache.rat.ui.UIOptionCollection;
37  import org.apache.rat.utils.DefaultLog;
38  
39  import edu.umd.cs.findbugs.annotations.SuppressFBWarnings;

41  /**
42   * Uses the AbstractOptionCollection to parse the command line options.
43   * Contains utility methods to ReportConfiguration from the options and an array of arguments.
44   *
45   * @param <T> The UIOption type that this parser is handling.
46   */
47  @SuppressFBWarnings("EI_EXPOSE_REP2")
48  public final class OptionCollectionParser<T extends UIOption<T>> {
49      /**
50       * The OptionCollection that we are working with.
51       */
52      private final UIOptionCollection<T> uiOptionCollection;
53  
54      /**
55       * Constructor.
56       * @param optionCollection the option collection to use for.
57       */
58      public OptionCollectionParser(final UIOptionCollection<T> optionCollection) {
59          this.uiOptionCollection = optionCollection;
60      }
61  
62      /**
63       * Parses the standard options to create a ReportConfiguration.
64       *
65       * @param workingDirectory the directory to resolve relative file names against.
66       * @param args the arguments to parse.
67       * @return the ArgumentContext for the process.
68       * @throws RatException on error.
69       */
70      public ArgumentContext parseCommands(final File workingDirectory, final String[] args)
71              throws RatException {
72          return parseCommands(workingDirectory, args, uiOptionCollection.getOptions());
73      }
74  
75      /**
76       * Parse the options into the command line.
77       * @param opts the option definitions.
78       * @param args the argument to apply the definitions to.
79       * @return the CommandLine
80       * @throws ParseException on option parsing error.
81       */
82      public static CommandLine parseCommandLine(final Options opts, final String[] args) throws ParseException {
83          try {
84              return DefaultParser.builder().setDeprecatedHandler(DeprecationReporter.getLogReporter())
85                      .setAllowPartialMatching(true).build().parse(opts, args);
86          } catch (ParseException e) {
87              DefaultLog.getInstance().error(e.getMessage());
88              DefaultLog.getInstance().error("Please use the \"--help\" option to see a list of valid commands and options.", e);
89              throw e;
90          }
91      }
92  
93      // visible for testing
94      void printHelp(final ArgumentContext argumentContext) throws RatException {
95          try {
96              new Licenses(argumentContext.getConfiguration(),
97                      new PrintWriter(argumentContext.getConfiguration().getOutput().get(),
98                              false, StandardCharsets.UTF_8)).printHelp();
99          } catch (IOException e) {
100             throw new RatException("Unable to print help: " + e.getMessage(), e);
101         }
102     }
103 
104     /**
105      * Parses the standard options to create a ReportConfiguration.
106      *
107      * @param workingDirectory The directory to resolve relative file names against.
108      * @param args the arguments to parse.
109      * @param options an Options object containing Apache command line options.
110      * @return the ArgumentContext for the process.
111      * @throws RatException on error.
112      */
113     // visible for testing
114     ArgumentContext parseCommands(final File workingDirectory, final String[] args,
115                                                                        final Options options) throws RatException {
116         try {
117             uiOptionCollection.resetSelected();
118             ArgumentContext argumentContext = new ArgumentContext(workingDirectory, options, args);
119             Arg.processLogLevel(argumentContext, uiOptionCollection);
120             populateConfiguration(argumentContext);
121             if (uiOptionCollection.isSelected(Arg.HELP_LICENSES)) {
122                 printHelp(argumentContext);
123             }
124             return argumentContext;
125         } catch (ParseException e) {
126             throw new RatException("Unable to parse command line: " + e.getMessage(), e);
127         }
128     }
129 
130     /**
131      * Create the report configuration.
132      * Note: this method is package private for testing.
133      * You probably want one of the {@code parseCommands(..)} methods.
134      * @param argumentContext The context to execute in.
135      * @return a ReportConfiguration
136      */
137     private ReportConfiguration populateConfiguration(final ArgumentContext argumentContext) {
138         argumentContext.processArgs(uiOptionCollection);
139         final ReportConfiguration configuration = argumentContext.getConfiguration();
140         if (!configuration.hasSource()) {
141             for (String s : argumentContext.getArgs()) {
142                 Reportable reportable = OptionCollection.getReportable(new File(s), configuration);
143                 if (reportable != null) {
144                     configuration.addSource(reportable);
145                 }
146             }
147         }
148         return configuration;
149     }
150 }
