19  package org.apache.rat.utils;
20  
21  import java.io.InputStream;
22  
23  import javax.xml.XMLConstants;
24  import javax.xml.transform.OutputKeys;
25  import javax.xml.transform.Transformer;
26  import javax.xml.transform.TransformerConfigurationException;
27  import javax.xml.transform.TransformerFactory;
28  import javax.xml.transform.stream.StreamSource;
29  
30  /**
31   * Factory to create standard XML objects. The intention of this class is to resolve in a consistent manner the
32   * XXE errors and similar XML IO errors.
33   */
34  public final class StandardXmlFactory {
35  
36      private StandardXmlFactory() {
37          // do not instantiate.
38      }
39      /**
40       * Create a transformer with no stylesheet.
41       * @return the transformer.
42       * @throws TransformerConfigurationException on error.
43       */
44      public static Transformer create() throws TransformerConfigurationException {
45          return create(null);
46      }
47  
48      /**
49       * Create a transformer with the specified stylesheet.
50       * @param styleIn the stylesheet to use.
51       * @return the transformer.
52       * @throws TransformerConfigurationException on error.
53       */
54      public static Transformer create(final InputStream styleIn) throws TransformerConfigurationException {
55          TransformerFactory factory = TransformerFactory.newInstance();
56          factory.setFeature(XMLConstants.FEATURE_SECURE_PROCESSING, true);
57          factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_DTD, "");
58          factory.setAttribute(XMLConstants.ACCESS_EXTERNAL_STYLESHEET, "");
59          Transformer transformer = styleIn == null ? factory.newTransformer() : factory.newTransformer(new StreamSource(styleIn));
60          transformer.setOutputProperty(OutputKeys.OMIT_XML_DECLARATION, "yes");
61          transformer.setOutputProperty(OutputKeys.METHOD, "xml");
62          transformer.setOutputProperty(OutputKeys.INDENT, "yes");
63          transformer.setOutputProperty(OutputKeys.ENCODING, "UTF-8");
64          transformer.setOutputProperty("{http://xml.apache.org/xslt}indent-amount", "4");
65          return transformer;
66      }
67  }
