47      private final static String[] DOS = { "-d-o-s", " 1990 - 1991 an owner" };

56      /* patterns are name, start, end, owner, pass results, fail results */
57      static final Arguments startStopOwner = Arguments.of("start-stop-owner", "1990", "1991", "an owner",
58              expandResults(DO, OD, DOS, ODS), expandResults(D, S, O, OS, SO));
59      static final Arguments startOwner = Arguments.of("start-owner", "1990", null, "an owner", expandResults(OS, SO, OD, ODS),
60              expandResults(D, DO, DOS, S, O));
61      static final Arguments start = Arguments.of("start", "1990", null, null, expandResults(D, DO, DOS, S, SO),
62              expandResults(OD, ODS, O, OS));
63      static final Arguments owner = Arguments.of("owner", null, null, "an owner", expandResults(DO, DOS, O, OD, ODS, OS, SO),
64              expandResults(S, D));
65      static final Arguments nada = Arguments.of("nada", null, null, null, expandResults(D, DO, DOS, S, SO),
66              expandResults(OD, ODS, O, OS));
67  
68      public static Stream<Arguments> parameterProvider() {
69          return Stream.of(startStopOwner, startOwner, start, owner, nada);
70      }

72      private static String[][] expandResults(String[]... args) {
73          List<String[]> arry = new ArrayList<>();
74          for (String pfx : prefix) {
75              Arrays.stream(args).map(origin -> new String[] { pfx + origin[0], pfx + origin[1] }).forEach(arry::add);
76          }
77          return arry.toArray(new String[arry.size()][2]);
78      }
79  
80      private static void verify(String testName, String[][] pass, String[][] fail) {
81          assertEquals(TOTAL_TESTS, pass.length + fail.length, "Wrong number of pass/fail tests");
82          Set<String> passSet = new HashSet<>();
83          Arrays.stream(pass).forEach(s -> passSet.add(s[0]));
84          Set<String> failSet = new HashSet<>();
85          Arrays.stream(fail).forEach(s -> failSet.add(s[0]));
86          for (String s : passSet) {
87              assertFalse(failSet.contains(s),
88                      () -> String.format("%s is in both pass and fail sets for %s", s, testName));
89          }
90      }
91  
92      @ParameterizedTest
93      @MethodSource("parameterProvider")
94      public void testPass(String testName, String start, String stop, String owner, String[][] pass, String[][] fail) {
95          verify(testName, pass, fail);
96          CopyrightMatcher matcher = new CopyrightMatcher(start, stop, owner);
97          for (String[] target : pass) {
98              IHeaders headers = AbstractMatcherTest.makeHeaders(target[TEXT], null);
99              assertTrue(matcher.matches(headers), () -> String.format("%s:%s on '%s' failed", testName, target[TEXT], target[TEXT]));
100             matcher.reset();
101         }
102     }
103 
104     @ParameterizedTest
105     @MethodSource("parameterProvider")
106     public void testFail(String testName, String start, String stop, String owner, String[][] pass, String[][] fail) {
107         verify(testName, pass, fail);
108         CopyrightMatcher matcher = new CopyrightMatcher(start, stop, owner);
109         for (String[] target : fail) {
110             IHeaders headers = AbstractMatcherTest.makeHeaders(target[TEXT], null);
111             assertFalse(matcher.matches(headers), String.format("%s:%s on '%s' passed", testName, target[NAME], target[TEXT]));
112             matcher.reset();
113         }
114     }
