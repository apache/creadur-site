19  package org.apache.rat.analysis;
20  
21  import java.io.BufferedReader;
22  import java.io.IOException;
23  import java.io.StringReader;
24  import java.util.ArrayList;
25  import java.util.List;
26  import java.util.stream.Stream;
27  
28  import org.apache.rat.configuration.builders.AnyBuilder;
29  import org.junit.jupiter.params.ParameterizedTest;
30  import org.junit.jupiter.params.provider.Arguments;
31  import org.junit.jupiter.params.provider.MethodSource;
32  
33  import static org.assertj.core.api.Assertions.assertThat;
34  
35  public class GeneratedFileTest {
36      private static final String[][] targets = {{"Cayenne", "generated by Cayenne"}, {"JJTree", "Generated By:JJTree"},
37              {"JavaCC", "Generated By:JavaCC"}, {"AUTOMATIC", "THIS FILE IS AUTOMATICALLY GENERATED"},
38              {"XBeans", "NOTE: this file is autogenerated by XBeans"},
39              {"automatic", "This file was automatically generated by"},
40              {"do not edit", "# WARNING: DO NOT EDIT OR DELETE THIS WORKSPACE FILE!"},
41              {"NMAKE", "# Microsoft Developer Studio Generated NMAKE File"},
42              {"MDS Build", "# Microsoft Developer Studio Generated Build File"},
43              {"autoheader", "Generated from configure.ac by autoheader"},
44              {"aclocal", "generated automatically by aclocal"},
45              {"maven project", "build.xml generated by maven from project.xml"},
46              {"generated", "This file was generated by"}, {"auto", "This file has been automatically generated."},
47              {"do not modify", "Automatically generated - do not modify!"}, {"javadoc style", "Javadoc style sheet"},
48              {"SOURCE", "SOURCE FILE GENERATED"}, {"Batik", "Generated by the Batik"},
49              {"file autogenerated", "this file is autogenerated"},
50              {"class autogenerated", "This class was autogenerated"}, {"maven", "Generated by Maven"},
51              {"Thrift", "Autogenerated by Thrift"},
52              {"machine generated", "DO NOT EDIT THIS FILE - it is machine generated"},
53              {"generated by", "This class was generated by"}, {"javadoc", "Generated by javadoc"}};
54  
55      public static Stream<Arguments> parameterProvider() {
56          List<Arguments> lst = new ArrayList<>();
57  
58          for (String[] target : targets) {
59              lst.add(Arguments.of(target[0], target[1]));
60          }
61          return lst.stream();
62      }
63  
64      private final IHeaderMatcher matcher;
65  
66      GeneratedFileTest() {
67          matcher = new AnyBuilder().setResource("/org/apache/rat/generation-keywords.txt").build();
68      }
69  
70      @ParameterizedTest
71      @MethodSource("parameterProvider")
72      public void testMatchProcessing(String name, String text) throws IOException {
73          assertThat(processText(text)).as(name).isTrue();
74      }
75  
76      private boolean processText(String text) throws IOException {
77          try (BufferedReader in = new BufferedReader(new StringReader(text))) {
78              IHeaders headers = HeaderCheckWorker.readHeader(in,
79                      HeaderCheckWorker.DEFAULT_NUMBER_OF_RETAINED_HEADER_LINES);
80              return matcher.matches(headers);
81          }
82      }
83  }
