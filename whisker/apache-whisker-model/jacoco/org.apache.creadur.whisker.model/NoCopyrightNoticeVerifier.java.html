package org.apache.creadur.whisker.model;

/**
 * Verifies that no copyright notices are present.
 */
public class NoCopyrightNoticeVerifier extends Visitor {
    /**
     * Set to true when a  {@link WithLicense} with
     * a copyright notice set is visited.
     */
    private boolean isCopyrightNoticePresent = false;

    /**
     * Visits a license to determine whether a copyright notice
     * has been set.
     * @param license not null
     */
    @Override
    public void visit(WithLicense license) {
        super.visit(license);
        if (license.hasCopyrightNotice()) {
            isCopyrightNoticePresent = true;
        }
    }

    /**
     * Has any {@link WithLicense} been visited
     * with a copyright notice set?
     * @return true when this verifier has visited a {@link WithLicense}
     * with a copyright notice set,
     * false otherwise
     */
    public boolean isCopyrightNoticePresent() {
        return isCopyrightNoticePresent;
    }
}
