48      public void processExampleFileTest() throws IOException {
49          try {
50              String[] lines = {
51                      "# somethings",
52                      "!thingone", "thing*", System.lineSeparator(),
53                      "# some fish",
54                      "**/fish", "*_fish",
55                      "# some colorful directories",
56                      "red/", "blue/*/"};
57              List<String> matches = Arrays.asList("some/things", "some/fish", "another/red_fish");
58  
59              List<String> notMatches = Arrays.asList("some/thingone", "thingone");
60  
61              writeFile(".gitignore", Arrays.asList(lines));
62  
63              assertCorrect(new GitIgnoreBuilder() {
64                  @Override
65                  protected Optional<File> globalGitIgnore() {
66                      return Optional.empty();
67                  }
68              }, matches, notMatches);
69          } finally {
70              System.getProperties().remove("FSInfo");
71          }
72      }

75      @ParameterizedTest
76      @MethodSource("modifyEntryData")
77      public void modifyEntryTest(DocumentName.FSInfo fsInfo, String source, String expected) {
78          GitIgnoreBuilder underTest = new GitIgnoreBuilder();
79          DocumentName testName = DocumentName.builder(fsInfo).setName("GitIgnoreBuilderTest").setBaseName("testDir").build();
80          List<MatcherSet> matcherSets = new ArrayList<>();
81          Optional<String> entry = underTest.modifyEntry(matcherSets::add, testName, source);
82  
83          if (source.endsWith("/")) {
84              assertThat(entry).isNotPresent();
85              assertThat(matcherSets).hasSize(1);
86              DocumentNameMatcher matcher = matcherSets.get(0).createMatcher();
87              assertThat(matcher.toString()).isEqualTo(expected);
88          } else {
89              assertThat(entry.get()).isEqualTo(expected);
90          }
91      }

93      private static Stream<Arguments> modifyEntryData() {
94          List<Arguments> lst = new ArrayList<>();
95          for (DocumentName.FSInfo fsInfo : FSInfoTest.TEST_SUITE) {
96              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
97  
98              lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
99              lst.add(Arguments.of(fsInfo, "\\#filename", "**/#filename"));
100             lst.add(Arguments.of(fsInfo, "!#filename", "!**/#filename"));
101             lst.add(Arguments.of(fsInfo, "/filename", "filename"));
102             lst.add(Arguments.of(fsInfo, "file/name", "file/name"));
103             lst.add(Arguments.of(fsInfo, "/file/name", "file/name"));
104             lst.add(Arguments.of(fsInfo, "filename", "**/filename"));
105             lst.add(Arguments.of(fsInfo, "filename/", "not(and(isDirectory, **/filename))"));
106             lst.add(Arguments.of(fsInfo, "/filename/", "not(and(isDirectory, filename))"));
107             // inclusion by itself becomes nothing.
108             lst.add(Arguments.of(fsInfo, "!filename/", "TRUE"));
109         }
110         return lst.stream();
111     }

113     private void assertMatches(DocumentName documentName, DocumentNameMatcher matcher, String[] matching, String[] notMatching) {
114         for (String test : matching) {
115             DocumentName name = documentName.resolve(test);
116             assertThat(matcher.matches(name)).as(test).isTrue();
117         }
118         for (String test : notMatching) {
119             DocumentName name = documentName.resolve(test);
120             assertThat(matcher.matches(name)).as(test).isFalse();
121         }
122     }

152      * see <a href="https://issues.apache.org/jira/browse/RAT-473">RAT-473</a> for details.

154     @Test
155     public void test_global_gitignore() throws URISyntaxException {

160                 String globalGitIgnore;
161                 try {
162                     globalGitIgnore = Paths.get(globalGitIgnoreUrl.toURI()).toString();
163                     return Optional.of(new File(globalGitIgnore));
164                 } catch (URISyntaxException e) {
165                     System.err.println("Unable to load global gitignore in test from " + globalGitIgnoreUrl + ", due to " + e.getMessage());
166                     return Optional.empty();
167                 }
168             }
169         };
170         URL url = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/src/");
171         File file = Paths.get(url.toURI()).toFile();
172 
173         DocumentName documentName = DocumentName.builder(file).build();

175         DocumentNameMatcher matcher = MatcherSet.merge(matcherSets).createMatcher();
176 
177         assertThat(matcher.toString()).isEqualTo("matcherSet(or('included dir1/.gitignore', 'included .gitignore', 'included global gitignore'), or('excluded dir1/.gitignore', **/.gitignore, 'excluded .gitignore', 'excluded global gitignore'))");
178 
179         // files that should be checked:
180         // "local-should-precede-global.md" should be 'matching'
181         // here, but that is a future improvement (RAT-476)
182         List<String> notMatching = Arrays.asList("dir1/dir1.md", "dir2/dir2.txt", "dir3/file3.log", "dir1/file1.log", "local-should-precede-global.md", "local-should-precede-global.xml");

186 
187         assertCorrect(matcherSets, documentName.getBaseDocumentName(), matching, notMatching);
188     }

191      * Test that exclusions from a global gitignore configured via
192      * {@code core.excludesFile} in a git configuration file are also applied,
193      * see <a href="https://issues.apache.org/jira/browse/RAT-578">RAT-578</a> for details.
194      */
195     @Test
196     public void test_global_gitignore_from_config() throws URISyntaxException, IOException {
197         URL globalGitIgnoreUrl = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/global-gitignore");
198         File globalGitIgnore = Paths.get(globalGitIgnoreUrl.toURI()).toFile();
199         String gitConfig = String.join(System.lineSeparator(),
200                 "[core]",
201                 "excludesFile = " + globalGitIgnore.getAbsolutePath());
202 
203         File configDir = tmpPath.toFile();
204         File userConfig = new File(configDir, ".gitconfig");
205         Files.createDirectories(configDir.toPath());
206         Files.write(userConfig.toPath(), Arrays.asList(gitConfig), StandardCharsets.UTF_8);
207 
208         GitIgnoreBuilder underTest = new GitIgnoreBuilder() {
209             @Override
210             protected Optional<File> globalGitIgnore() {
211                 // force reading the config file from the test directory
212                 GitConfig gitConfig = new GitConfig(null, userConfig, tmpPath.toString(), tmpPath.toFile());
213                 return gitConfig.coreExcludesFile();
214             }
215         };
216         URL url = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/src/");
217         File file = Paths.get(url.toURI()).toFile();
218 
219         DocumentName documentName = DocumentName.builder(file).build();
220         List<MatcherSet> matcherSets = underTest.build(documentName);
221         DocumentNameMatcher matcher = MatcherSet.merge(matcherSets).createMatcher();
222 
223         assertThat(matcher.toString()).isEqualTo("matcherSet(or('included dir1/.gitignore', 'included .gitignore', 'included global gitignore'), or('excluded dir1/.gitignore', **/.gitignore, 'excluded .gitignore', 'excluded global gitignore'))");
224 
225         // files that should be checked:
226         List<String> notMatching = Arrays.asList("dir1/dir1.md", "dir2/dir2.txt", "dir3/file3.log", "dir1/file1.log", "local-should-precede-global.md", "local-should-precede-global.xml");
227 
228         // files that should be ignored:
229         List<String> matching = Arrays.asList(".gitignore", "README.txt", "root.md", "dir1/.gitignore", "dir1/dir1.txt", "dir2/dir2.md", "dir3/dir3.log");

234     /**
235      * Test that a relative {@code core.excludesFile} is resolved once against the scan root and
236      * NOT re-resolved against a subdirectory when the scan descends (see
237      * <a href="https://github.com/apache/creadur-rat/pull/746">PR #746</a> review).
238      * <p>Mirrors git semantics where patterns from {@code core.excludesFile} are treated as if
239      * specified at the root of the working tree: a same-named file deeper in the tree must not be
240      * picked up.</p>
241      */
242     @Test
243     public void test_relative_excludesFile_resolved_once_at_root() throws URISyntaxException {
244         URL srcUrl = GitIgnoreBuilderTest.class.getClassLoader().getResource("GitIgnoreBuilderTest/RAT-578-root/");
245         File src = Paths.get(srcUrl.toURI()).toFile();
246 
247         // simulate a core.excludesFile relative to the scan root (CWD), matching git's resolution
248         File resolved = new File(src, "excludes.txt");
249         GitIgnoreBuilder underTest = new GitIgnoreBuilder() {
250             @Override
251             protected Optional<File> globalGitIgnore() {
252                 return Optional.of(resolved);
253             }
254         };
255 
256         DocumentName documentName = DocumentName.builder(src).build();
257         List<MatcherSet> matcherSets = underTest.build(documentName);
258 
259         // excluded by the root excludes.txt (*.md)
260         List<String> matching = Arrays.asList("root.md", "dir/ignored.md");
261         // dir/excludes.txt (*.log) is a decoy and must NOT be used:
262         // dir/kept.log stays checked instead of being ignored
263         List<String> notMatching = Arrays.asList("dir/kept.log", "dir/kept.txt", "dir/README.md");
264 
265         assertCorrect(matcherSets, documentName.getBaseDocumentName(), matching, notMatching);
266     }

268     /**
269      * A {@code false} value for {@code RAT_GIT_GLOBAL_IGNORE} disables the global git ignore so that

279     /**
280      * An explicit path for {@code RAT_GIT_GLOBAL_IGNORE} uses that file deterministically instead of
281      * the machine's git configuration.
282      */
283     @Test
284     public void test_global_git_ignore_explicit_path() throws IOException {
285         File ignore = tmpPath.resolve("global-ignore").toFile();
286         Files.write(ignore.toPath(), Arrays.asList("*.log"), StandardCharsets.UTF_8);
287 
288         assertThat(GitIgnoreBuilder.resolveGlobalGitIgnore(ignore.getAbsolutePath()))
289                 .hasValue(ignore);
290         // a path that does not exist yields no global ignore (no fall back to the git config)
291         assertThat(GitIgnoreBuilder.resolveGlobalGitIgnore(ignore.getAbsolutePath() + ".missing"))
292                 .isEmpty();
293     }

295     /**
296      * A {@code null} or {@code true} value keeps the git-configuration resolution (RAT-578 default).

298     @Test
